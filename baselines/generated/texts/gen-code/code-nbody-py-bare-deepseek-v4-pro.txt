#!/usr/bin/env python3
"""
N-Body Gravitational Simulation with Leapfrog Integrator using NumPy.

This module provides a complete, self-contained N-body simulation. It initializes
a random set of particles, integrates their motion using the leapfrog (kick-drift-kick)
scheme, and optionally plots the resulting trajectories.

Usage:
    python nbody_leapfrog.py [--bodies N] [--steps S] [--dt DT] [--plot]
"""

import argparse
import sys
from typing import Tuple

import numpy as np

# Physical constants
G = 1.0  # gravitational constant
SOFTENING = 1e-3  # softening length to avoid singularities


def gravitational_acceleration(
    positions: np.ndarray, masses: np.ndarray, G: float = G, softening: float = SOFTENING
) -> np.ndarray:
    """
    Compute the gravitational acceleration for each particle.

    Args:
        positions: (N, 3) array of positions.
        masses: (N,) array of masses.
        G: Gravitational constant.
        softening: Softening length.

    Returns:
        (N, 3) array of accelerations.
    """
    N = positions.shape[0]
    acc = np.zeros_like(positions)

    for i in range(N):
        # Vector from all other particles to particle i
        r_vec = positions[i] - positions  # (N, 3)
        # Squared distances with softening
        r2 = np.sum(r_vec**2, axis=1) + softening**2
        # Inverse distance cubed
        r_inv3 = r2 ** (-1.5)
        # Mass factor (we don't want self-interaction)
        m_factor = masses * r_inv3
        m_factor[i] = 0.0  # remove self-contribution
        # Accumulate acceleration
        acc[i] = -G * np.sum(r_vec * m_factor[:, np.newaxis], axis=0)

    return acc


def leapfrog_step(
    positions: np.ndarray,
    velocities: np.ndarray,
    accelerations: np.ndarray,
    masses: np.ndarray,
    dt: float,
    G: float = G,
    softening: float = SOFTENING,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Advance the system by one time step using the leapfrog (kick-drift-kick) scheme.

    Args:
        positions: (N, 3) current positions.
        velocities: (N, 3) current velocities (at half-step if continuing).
        accelerations: (N, 3) current accelerations (at t).
        masses: (N,) masses.
        dt: Time step.
        G: Gravitational constant.
        softening: Softening length.

    Returns:
        Updated positions, velocities, and accelerations.
    """
    # Kick: half-step velocity update
    velocities += 0.5 * dt * accelerations

    # Drift: full-step position update
    positions += dt * velocities

    # Recompute accelerations at new positions
    accelerations = gravitational_acceleration(positions, masses, G, softening)

    # Kick: second half-step velocity update
    velocities += 0.5 * dt * accelerations

    return positions, velocities, accelerations


def initialize_system(
    N: int, seed: int = 42, max_radius: float = 5.0, max_mass: float = 1.0
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """
    Initialize a random N-body system.

    Positions and velocities are drawn from normal distributions,
    and masses are uniformly distributed.

    Args:
        N: Number of particles.
        seed: Random seed for reproducibility.
        max_radius: Standard deviation of initial positions.
        max_mass: Maximum mass of a particle.

    Returns:
        positions, velocities, accelerations, masses
    """
    rng = np.random.default_rng(seed)

    positions = rng.normal(0, max_radius, size=(N, 3))
    velocities = rng.normal(0, max_radius * 0.2, size=(N, 3))  # slower velocities
    masses = rng.uniform(0.1, max_mass, size=N)

    # Center-of-mass correction: zero out mean velocity
    total_mass = np.sum(masses)
    vel_cm = np.sum(velocities * masses[:, np.newaxis], axis=0) / total_mass
    velocities -= vel_cm

    # Compute initial accelerations
    accelerations = gravitational_acceleration(positions, masses)

    return positions, velocities, accelerations, masses


def simulate(
    positions: np.ndarray,
    velocities: np.ndarray,
    masses: np.ndarray,
    dt: float,
    steps: int,
    store_every: int = 1,
    progress: bool = True,
) -> Tuple[np.ndarray, np.ndarray]:
    """
    Run the N-body simulation.

    Args:
        positions: Initial positions (N, 3).
        velocities: Initial velocities (N, 3).
        masses: Masses (N,).
        dt: Time step size.
        steps: Number of simulation steps.
        store_every: Store trajectory every this many steps.
        progress: Print progress bar.

    Returns:
        pos_history: (frames, N, 3) position history.
        energy_history: (frames,) total energy history (kinetic + potential).
    """
    N = positions.shape[0]
    store_count = steps // store_every
    pos_history = np.zeros((store_count + 1, N, 3))
    energy_history = np.zeros(store_count + 1)

    # Initial accelerations
    acc = gravitational_acceleration(positions, masses)

    def compute_energy(pos, vel, mass):
        """Compute total kinetic + potential energy."""
        kinetic = 0.5 * np.sum(mass * np.sum(vel**2, axis=1))
        # Potential energy
        pot = 0.0
        for i in range(len(mass)):
            r_vec = pos[i] - pos
            r2 = np.sum(r_vec**2, axis=1) + SOFTENING**2
            r = np.sqrt(r2)
            # Avoid double counting and self contribution
            m_mask = masses.copy()
            m_mask[i] = 0.0
            pot += -G * mass[i] * np.sum(m_mask / r)
        pot *= 0.5  # each pair counted twice
        return kinetic + pot

    # Store initial state
    pos_history[0] = positions.copy()
    energy_history[0] = compute_energy(positions, velocities, masses)

    if progress:
        print("Running simulation...")

    for step in range(1, steps + 1):
        # Advance one step
        positions, velocities, acc = leapfrog_step(
            positions, velocities, acc, masses, dt
        )

        if step % store_every == 0:
            idx = step // store_every
            pos_history[idx] = positions.copy()
            energy_history[idx] = compute_energy(positions, velocities, masses)

        if progress and step % max(1, steps // 10) == 0:
            pct = step / steps * 100
            print(f"  Progress: {pct:.0f}%")

    if progress:
        print("Simulation complete.")

    return pos_history, energy_history


def plot_results(
    pos_history: np.ndarray,
    energy_history: np.ndarray,
    dt: float,
    store_every: int,
) -> None:
    """
    Plot trajectories and energy conservation using matplotlib.
    """
    try:
        import matplotlib.pyplot as plt
        from mpl_toolkits.mplot3d import Axes3D  # noqa: F401
    except ImportError:
        print("matplotlib not installed. Skipping plot.", file=sys.stderr)
        return

    fig = plt.figure(figsize=(12, 5))

    # 3D trajectory plot
    ax = fig.add_subplot(1, 2, 1, projection="3d")
    N = pos_history.shape[1]
    for i in range(N):
        ax.plot(
            pos_history[:, i, 0],
            pos_history[:, i, 1],
            pos_history[:, i, 2],
            alpha=0.7,
        )
    ax.set_title("Particle Trajectories")
    ax.set_xlabel("X")
    ax.set_ylabel("Y")
    ax.set_zlabel("Z")

    # Energy plot
    ax2 = fig.add_subplot(1, 2, 2)
    times = np.arange(len(energy_history)) * (dt * store_every)
    energy_rel = (energy_history - energy_history[0]) / abs(energy_history[0])
    ax2.plot(times, energy_rel)
    ax2.set_title("Relative Energy Error")
    ax2.set_xlabel("Time")
    ax2.set_ylabel("ΔE / |E₀|")
    ax2.grid(True)

    plt.tight_layout()
    plt.show()


def main() -> None:
    parser = argparse.ArgumentParser(
        description="N-body gravitational simulation with leapfrog integrator."
    )
    parser.add_argument(
        "--bodies",
        type=int,
        default=20,
        help="Number of bodies (default: 20)",
    )
    parser.add_argument(
        "--steps",
        type=int,
        default=2000,
        help="Number of simulation steps (default: 2000)",
    )
    parser.add_argument(
        "--dt",
        type=float,
        default=0.01,
        help="Time step size (default: 0.01)",
    )
    parser.add_argument(
        "--plot",
        action="store_true",
        default=False,
        help="Plot trajectories and energy error after simulation.",
    )
    parser.add_argument(
        "--seed",
        type=int,
        default=42,
        help="Random seed (default: 42).",
    )
    parser.add_argument(
        "--store-every",
        type=int,
        default=10,
        help="Store frame every N steps (default: 10).",
    )

    args = parser.parse_args()

    # Initialize system
    print(f"Initializing {args.bodies} bodies...")
    positions, velocities, accelerations, masses = initialize_system(
        N=args.bodies, seed=args.seed
    )
    print(f"Total mass: {np.sum(masses):.2f}")

    # Run simulation
    pos_history, energy_history = simulate(
        positions,
        velocities,
        masses,
        dt=args.dt,
        steps=args.steps,
        store_every=args.store_every,
    )

    # Print energy conservation stats
    energy_error = np.abs((energy_history[-1] - energy_history[0]) / energy_history[0])
    print(f"Final relative energy error: {energy_error:.6e}")

    # Plot if requested
    if args.plot:
        plot_results(pos_history, energy_history, args.dt, args.store_every)


if __name__ == "__main__":
    main()

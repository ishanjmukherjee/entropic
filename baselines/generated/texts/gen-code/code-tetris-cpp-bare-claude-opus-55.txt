// blockfall.cpp
//
// Core logic for a falling-block puzzle game, with no rendering.
//
// Features:
//   * Configurable board with hidden spawn rows above the visible field
//   * Seven standard tetrominoes with Super Rotation System (SRS) wall kicks
//   * Collision detection, ghost piece and line clearing
//   * 7-bag randomizer with a configurable preview queue
//   * Hold, soft drop and hard drop
//   * Time-based gravity and lock delay with a capped number of move resets
//   * Guideline-style scoring: T-spins (3-corner rule), back-to-back,
//     combos and perfect clears
//   * Block-out and lock-out game-over detection
//
// Coordinates: x grows to the right and y grows downward. Row 0 is the top
// of the buffer zone. Rows [0, hiddenRows) are hidden spawn rows. Rows
// [hiddenRows, hiddenRows + visibleHeight) form the visible playfield.
//
// Build the self-test with: g++ -std=c++17 -O2 -DBLOCKFALL_SELFTEST blockfall.cpp

#include <algorithm>
#include <array>
#include <cmath>
#include <cstdint>
#include <deque>
#include <optional>
#include <random>
#include <stdexcept>
#include <string>
#include <utility>
#include <vector>

namespace blockfall {

// ---------------------------------------------------------------------------
// Basic types
// ---------------------------------------------------------------------------

enum class PieceType : std::uint8_t { I = 0, O, T, S, Z, J, L };
inline constexpr int kPieceTypeCount = 7;

enum class Rotation : std::uint8_t { Spawn = 0, Right = 1, Reverse = 2, Left = 3 };

constexpr Rotation rotatedCW(Rotation r) noexcept {
    return static_cast<Rotation>((static_cast<int>(r) + 1) & 3);
}
constexpr Rotation rotatedCCW(Rotation r) noexcept {
    return static_cast<Rotation>((static_cast<int>(r) + 3) & 3);
}

struct Point {
    int x = 0;
    int y = 0;
};

constexpr bool operator==(Point a, Point b) noexcept { return a.x == b.x && a.y == b.y; }
constexpr bool operator!=(Point a, Point b) noexcept { return !(a == b); }

using Minos = std::array<Point, 4>;

namespace detail {

struct ShapeDef {
    int boxSize;   // side length of the rotation bounding box
    Minos spawn;   // mino offsets in the spawn orientation, box-relative
};

// Spawn orientations as defined by SRS (y down, box-relative).
inline constexpr std::array<ShapeDef, kPieceTypeCount> kShapes = {{
    /* I */ {4, {{{0, 1}, {1, 1}, {2, 1}, {3, 1}}}},
    /* O */ {4, {{{1, 0}, {2, 0}, {1, 1}, {2, 1}}}},
    /* T */ {3, {{{1, 0}, {0, 1}, {1, 1}, {2, 1}}}},
    /* S */ {3, {{{1, 0}, {2, 0}, {0, 1}, {1, 1}}}},
    /* Z */ {3, {{{0, 0}, {1, 0}, {1, 1}, {2, 1}}}},
    /* J */ {3, {{{0, 0}, {0, 1}, {1, 1}, {2, 1}}}},
    /* L */ {3, {{{2, 0}, {0, 1}, {1, 1}, {2, 1}}}},
}};

using RotationTable = std::array<std::array<Minos, 4>, kPieceTypeCount>;

// Rotating clockwise inside an N x N box with y down maps (x, y) to
// (N - 1 - y, x). This reproduces the SRS orientations exactly. The O piece
// does not change shape when rotated.
constexpr RotationTable buildRotationTable() {
    RotationTable table{};
    for (int t = 0; t < kPieceTypeCount; ++t) {
        const ShapeDef& def = kShapes[static_cast<std::size_t>(t)];
        table[static_cast<std::size_t>(t)][0] = def.spawn;
        for (int r = 1; r < 4; ++r) {
            for (int i = 0; i < 4; ++i) {
                const Point prev = table[static_cast<std::size_t>(t)][static_cast<std::size_t>(r - 1)]
                                        [static_cast<std::size_t>(i)];
                Point next = prev;
                if (t != static_cast<int>(PieceType::O)) {
                    next = Point{def.boxSize - 1 - prev.y, prev.x};
                }
                table[static_cast<std::size_t>(t)][static_cast<std::size_t>(r)]
                     [static_cast<std::size_t>(i)] = next;
            }
        }
    }
    return table;
}

inline constexpr RotationTable kRotationTable = buildRotationTable();

// SRS wall kick data, written in the conventional y-up notation.
// The y component is negated when applied, because this engine uses y down.
// Index: [fromRotation][direction], where direction 0 = CW and 1 = CCW.
using KickSet = std::array<Point, 5>;
using KickTable = std::array<std::array<KickSet, 2>, 4>;

inline constexpr KickTable kJlstzKicks = {{
    /* from 0 */ {{ {{{0, 0}, {-1, 0}, {-1,  1}, {0, -2}, {-1, -2}}},     // 0 -> R
                    {{{0, 0}, { 1, 0}, { 1,  1}, {0, -2}, { 1, -2}}} }},  // 0 -> L
    /* from R */ {{ {{{0, 0}, { 1, 0}, { 1, -1}, {0,  2}, { 1,  2}}},     // R -> 2
                    {{{0, 0}, { 1, 0}, { 1, -1}, {0,  2}, { 1,  2}}} }},  // R -> 0
    /* from 2 */ {{ {{{0, 0}, { 1, 0}, { 1,  1}, {0, -2}, { 1, -2}}},     // 2 -> L
                    {{{0, 0}, {-1, 0}, {-1,  1}, {0, -2}, {-1, -2}}} }},  // 2 -> R
    /* from L */ {{ {{{0, 0}, {-1, 0}, {-1, -1}, {0,  2}, {-1,  2}}},     // L -> 0
                    {{{0, 0}, {-1, 0}, {-1, -1}, {0,  2}, {-1,  2}}} }},  // L -> 2
}};

inline constexpr KickTable kIKicks = {{
    /* from 0 */ {{ {{{0, 0}, {-2, 0}, { 1, 0}, {-2, -1}, { 1,  2}}},     // 0 -> R
                    {{{0, 0}, {-1, 0}, { 2, 0}, {-1,  2}, { 2, -1}}} }},  // 0 -> L
    /* from R */ {{ {{{0, 0}, {-1, 0}, { 2, 0}, {-1,  2}, { 2, -1}}},     // R -> 2
                    {{{0, 0}, { 2, 0}, {-1, 0}, { 2,  1}, {-1, -2}}} }},  // R -> 0
    /* from 2 */ {{ {{{0, 0}, { 2, 0}, {-1, 0}, { 2,  1}, {-1, -2}}},     // 2 -> L
                    {{{0, 0}, { 1, 0}, {-2, 0}, { 1, -2}, {-2,  1}}} }},  // 2 -> R
    /* from L */ {{ {{{0, 0}, { 1, 0}, {-2, 0}, { 1, -2}, {-2,  1}}},     // L -> 0
                    {{{0, 0}, {-2, 0}, { 1, 0}, {-2, -1}, { 1,  2}}} }},  // L -> 2
}};

constexpr std::size_t idx(PieceType t) noexcept { return static_cast<std::size_t>(t); }
constexpr std::size_t idx(Rotation r) noexcept { return static_cast<std::size_t>(r); }

}  // namespace detail

constexpr int boxSize(PieceType type) noexcept {
    return detail::kShapes[detail::idx(type)].boxSize;
}

constexpr const Minos& shapeOf(PieceType type, Rotation rot) noexcept {
    return detail::kRotationTable[detail::idx(type)][detail::idx(rot)];
}

// ---------------------------------------------------------------------------
// Piece
// ---------------------------------------------------------------------------

struct Piece {
    PieceType type = PieceType::I;
    Rotation rotation = Rotation::Spawn;
    int x = 0;  // board column of the bounding box's left edge
    int y = 0;  // board row of the bounding box's top edge

    Minos cells() const noexcept {
        Minos out = shapeOf(type, rotation);
        for (Point& p : out) {
            p.x += x;
            p.y += y;
        }
        return out;
    }

    Piece translated(int dx, int dy) const noexcept {
        Piece p = *this;
        p.x += dx;
        p.y += dy;
        return p;
    }
};

// ---------------------------------------------------------------------------
// Board
// ---------------------------------------------------------------------------

class Board {
public:
    // 0 means empty. Any other value is (PieceType + 1).
    using Cell = std::uint8_t;
    static constexpr Cell kEmpty = 0;

    static constexpr Cell cellFor(PieceType t) noexcept {
        return static_cast<Cell>(static_cast<int>(t) + 1);
    }
    static constexpr std::optional<PieceType> typeOf(Cell c) noexcept {
        if (c == kEmpty || c > kPieceTypeCount) return std::nullopt;
        return static_cast<PieceType>(c - 1);
    }

    Board(int width, int height) : width_(width), height_(height) {
        if (width <= 0 || height <= 0) {
            throw std::invalid_argument("Board dimensions must be positive");
        }
        cells_.assign(static_cast<std::size_t>(width) * static_cast<std::size_t>(height), kEmpty);
    }

    int width() const noexcept { return width_; }
    int height() const noexcept { return height_; }

    bool inBounds(int x, int y) const noexcept {
        return x >= 0 && x < width_ && y >= 0 && y < height_;
    }

    Cell at(int x, int y) const {
        if (!inBounds(x, y)) throw std::out_of_range("Board::at out of range");
        return cells_[index(x, y)];
    }

    void set(int x, int y, Cell value) {
        if (!inBounds(x, y)) throw std::out_of_range("Board::set out of range");
        cells_[index(x, y)] = value;
    }

    // Cells outside the board count as solid. This makes walls and the floor
    // block movement.
    bool isBlocked(int x, int y) const noexcept {
        return !inBounds(x, y) || cells_[index(x, y)] != kEmpty;
    }

    bool fits(const Piece& piece) const noexcept {
        for (const Point& p : piece.cells()) {
            if (isBlocked(p.x, p.y)) return false;
        }
        return true;
    }

    // Writes the piece into the board. The caller must ensure it fits.
    void place(const Piece& piece) {
        const Cell value = cellFor(piece.type);
        for (const Point& p : piece.cells()) {
            if (!inBounds(p.x, p.y)) {
                throw std::logic_error("Board::place: piece is out of bounds");
            }
            cells_[index(p.x, p.y)] = value;
        }
    }

    bool isRowFull(int y) const noexcept {
        const auto begin = cells_.begin() + static_cast<std::ptrdiff_t>(index(0, y));
        return std::none_of(begin, begin + width_, [](Cell c) { return c == kEmpty; });
    }

    bool isRowEmpty(int y) const noexcept {
        const auto begin = cells_.begin() + static_cast<std::ptrdiff_t>(index(0, y));
        return std::all_of(begin, begin + width_, [](Cell c) { return c == kEmpty; });
    }

    std::vector<int> fullRows() const {
        std::vector<int> rows;
        for (int y = 0; y < height_; ++y) {
            if (isRowFull(y)) rows.push_back(y);
        }
        return rows;
    }

    // Removes every full row and lets the rows above fall into the gaps.
    // Returns the number of rows removed. Runs in O(width * height).
    int clearFullRows() {
        int write = height_ - 1;
        int cleared = 0;
        for (int read = height_ - 1; read >= 0; --read) {
            if (isRowFull(read)) {
                ++cleared;
                continue;
            }
            if (write != read) {
                std::copy_n(cells_.begin() + static_cast<std::ptrdiff_t>(index(0, read)), width_,
                            cells_.begin() + static_cast<std::ptrdiff_t>(index(0, write)));
            }
            --write;
        }
        for (int y = write; y >= 0; --y) {
            std::fill_n(cells_.begin() + static_cast<std::ptrdiff_t>(index(0, y)), width_, kEmpty);
        }
        return cleared;
    }

    bool isEmpty() const noexcept {
        return std::all_of(cells_.begin(), cells_.end(), [](Cell c) { return c == kEmpty; });
    }

    void clear() noexcept { std::fill(cells_.begin(), cells_.end(), kEmpty); }

private:
    std::size_t index(int x, int y) const noexcept {
        return static_cast<std::size_t>(y) * static_cast<std::size_t>(width_) +
               static_cast<std::size_t>(x);
    }

    int width_;
    int height_;
    std::vector<Cell> cells_;
};

// ---------------------------------------------------------------------------
// Randomizer: the 7-bag
// ---------------------------------------------------------------------------

class BagRandomizer {
public:
    explicit BagRandomizer(std::uint32_t seed) : rng_(seed) {}

    void reseed(std::uint32_t seed) {
        rng_.seed(seed);
        queue_.clear();
    }

    PieceType next() {
        ensure(1);
        const PieceType t = queue_.front();
        queue_.pop_front();
        return t;
    }

    // Returns the next `count` pieces without consuming them.
    std::vector<PieceType> peek(std::size_t count) {
        ensure(count);
        return std::vector<PieceType>(queue_.begin(),
                                      queue_.begin() + static_cast<std::ptrdiff_t>(count));
    }

private:
    void ensure(std::size_t count) {
        while (queue_.size() < count) refill();
    }

    void refill() {
        std::array<PieceType, kPieceTypeCount> bag = {PieceType::I, PieceType::O, PieceType::T,
                                                      PieceType::S, PieceType::Z, PieceType::J,
                                                      PieceType::L};
        std::shuffle(bag.begin(), bag.end(), rng_);
        queue_.insert(queue_.end(), bag.begin(), bag.end());
    }

    std::mt19937 rng_;
    std::deque<PieceType> queue_;
};

// ---------------------------------------------------------------------------
// Game
// ---------------------------------------------------------------------------

enum class TSpin : std::uint8_t { None, Mini, Full };
enum class GameOverReason : std::uint8_t { None, BlockOut, LockOut };

struct LockResult {
    PieceType piece = PieceType::I;
    int linesCleared = 0;
    TSpin tspin = TSpin::None;
    bool backToBack = false;    // the back-to-back bonus was applied
    int combo = -1;             // -1 means no active combo
    bool perfectClear = false;
    std::int64_t pointsAwarded = 0;
};

struct GameConfig {
    int width = 10;
    int visibleHeight = 20;
    int hiddenRows = 20;              // buffer rows above the visible field (>= 2)
    int previewCount = 5;
    bool holdEnabled = true;
    int startLevel = 1;
    int maxLevel = 30;
    int linesPerLevel = 10;
    double lockDelayMs = 500.0;
    int maxLockResets = 15;
    double softDropFactor = 20.0;
    std::optional<std::uint32_t> seed;  // nullopt means a nondeterministic seed
};

class Game {
public:
    explicit Game(GameConfig config = {})
        : config_(validate(std::move(config))),
          board_(config_.width, config_.visibleHeight + config_.hiddenRows),
          randomizer_(resolveSeed(config_.seed)) {
        resetState();
    }

    // Restarts the game. If a seed is given, it replaces the configured one.
    void reset(std::optional<std::uint32_t> seed = std::nullopt) {
        if (seed) config_.seed = seed;
        randomizer_.reseed(resolveSeed(config_.seed));
        resetState();
    }

    // ---- Input -----------------------------------------------------------

    bool moveLeft() { return shift(-1); }
    bool moveRight() { return shift(+1); }
    bool rotateCW() { return rotate(true); }
    bool rotateCCW() { return rotate(false); }

    // Moves the piece down one row and awards one point. Returns false if the
    // piece is grounded. This step does not lock the piece; lock delay still
    // applies.
    bool softDrop() {
        if (!canAct()) return false;
        if (!tryMove(0, 1)) return false;
        score_ += 1;
        lastMoveWasRotation_ = false;
        noteDescent();
        return true;
    }

    // While active, gravity runs softDropFactor times faster. Each row
    // dropped by gravity during this time awards one point.
    void setSoftDropActive(bool active) noexcept { softDropActive_ = active; }

    // Drops the piece to the floor and locks it immediately.
    // Returns the number of rows dropped.
    int hardDrop() {
        if (!canAct()) return 0;
        int rows = 0;
        while (tryMove(0, 1)) ++rows;
        if (rows > 0) lastMoveWasRotation_ = false;
        score_ += 2 * static_cast<std::int64_t>(rows);
        lockPiece();
        return rows;
    }

    bool hold() {
        if (!canAct() || !config_.holdEnabled || !canHold_) return false;
        const PieceType current = active_->type;
        if (held_) {
            const PieceType swapIn = *held_;
            held_ = current;
            spawn(swapIn);
        } else {
            held_ = current;
            spawn(randomizer_.next());
        }
        canHold_ = false;
        return true;
    }

    // Advances the simulation by dtMs milliseconds. This applies gravity and
    // lock delay.
    void update(double dtMs) {
        if (!canAct() || !(dtMs > 0.0)) return;

        const double interval = gravityIntervalMs();
        gravityAccumMs_ += dtMs;

        // Cap the number of rows per update. Very fast gravity cannot loop
        // forever and cannot do more work than the board height allows.
        int safety = board_.height();
        while (gravityAccumMs_ >= interval && safety-- > 0) {
            if (!tryMove(0, 1)) {
                gravityAccumMs_ = 0.0;
                break;
            }
            gravityAccumMs_ -= interval;
            if (softDropActive_) score_ += 1;
            lastMoveWasRotation_ = false;
            noteDescent();
        }
        if (safety <= 0) gravityAccumMs_ = 0.0;

        if (isGrounded()) {
            lockTimerMs_ += dtMs;
            if (lockTimerMs_ >= config_.lockDelayMs || lockResets_ >= config_.maxLockResets) {
                lockPiece();
            }
        } else {
            lockTimerMs_ = 0.0;
        }
    }

    // ---- Queries ---------------------------------------------------------

    const GameConfig& config() const noexcept { return config_; }
    const Board& board() const noexcept { return board_; }
    const std::optional<Piece>& activePiece() const noexcept { return active_; }
    std::optional<PieceType> heldPiece() const noexcept { return held_; }
    bool canHold() const noexcept { return config_.holdEnabled && canHold_; }
    std::vector<PieceType> preview() {
        return randomizer_.peek(static_cast<std::size_t>(config_.previewCount));
    }

    std::optional<Piece> ghostPiece() const {
        if (!active_) return std::nullopt;
        Piece g = *active_;
        while (board_.fits(g.translated(0, 1))) ++g.y;
        return g;
    }

    std::int64_t score() const noexcept { return score_; }
    int lines() const noexcept { return lines_; }
    int level() const noexcept { return level_; }
    int combo() const noexcept { return combo_; }
    bool backToBackActive() const noexcept { return backToBack_; }
    bool isGameOver() const noexcept { return gameOver_ != GameOverReason::None; }
    GameOverReason gameOverReason() const noexcept { return gameOver_; }
    const std::optional<LockResult>& lastLock() const noexcept { return lastLock_; }
    int piecesLocked() const noexcept { return piecesLocked_; }

    // The first visible row index on the board.
    int visibleTop() const noexcept { return config_.hiddenRows; }

    bool isGrounded() const noexcept {
        return active_ && !board_.fits(active_->translated(0, 1));
    }

    // Time per row in milliseconds under normal gravity. This follows the
    // guideline formula and is capped at level 20.
    double baseGravityIntervalMs() const noexcept {
        const int lvl = std::clamp(level_, 1, 20);
        const double base = 0.8 - static_cast<double>(lvl - 1) * 0.007;
        return std::pow(base, static_cast<double>(lvl - 1)) * 1000.0;
    }

private:
    static GameConfig validate(GameConfig c) {
        if (c.width < 4) throw std::invalid_argument("width must be at least 4");
        if (c.visibleHeight < 1) throw std::invalid_argument("visibleHeight must be positive");
        if (c.hiddenRows < 2) throw std::invalid_argument("hiddenRows must be at least 2");
        if (c.previewCount < 0) throw std::invalid_argument("previewCount must be non-negative");
        if (c.startLevel < 1) throw std::invalid_argument("startLevel must be at least 1");
        if (c.maxLevel < c.startLevel) throw std::invalid_argument("maxLevel < startLevel");
        if (c.linesPerLevel < 1) throw std::invalid_argument("linesPerLevel must be positive");
        if (!(c.lockDelayMs >= 0.0)) throw std::invalid_argument("lockDelayMs must be >= 0");
        if (c.maxLockResets < 0) throw std::invalid_argument("maxLockResets must be >= 0");
        if (!(c.softDropFactor >= 1.0)) throw std::invalid_argument("softDropFactor must be >= 1");
        return c;
    }

    static std::uint32_t resolveSeed(const std::optional<std::uint32_t>& seed) {
        if (seed) return *seed;
        std::random_device rd;
        return rd();
    }

    void resetState() {
        board_.clear();
        active_.reset();
        held_.reset();
        canHold_ = true;
        score_ = 0;
        lines_ = 0;
        level_ = config_.startLevel;
        combo_ = -1;
        backToBack_ = false;
        gameOver_ = GameOverReason::None;
        lastLock_.reset();
        piecesLocked_ = 0;
        softDropActive_ = false;
        spawn(randomizer_.next());
    }

    bool canAct() const noexcept { return !isGameOver() && active_.has_value(); }

    double gravityIntervalMs() const noexcept {
        const double base = baseGravityIntervalMs();
        return softDropActive_ ? base / config_.softDropFactor : base;
    }

    bool tryMove(int dx, int dy) {
        const Piece moved = active_->translated(dx, dy);
        if (!board_.fits(moved)) return false;
        active_ = moved;
        return true;
    }

    bool shift(int dx) {
        if (!canAct()) return false;
        if (!tryMove(dx, 0)) return false;
        lastMoveWasRotation_ = false;
        onSuccessfulManipulation();
        return true;
    }

    bool rotate(bool clockwise) {
        if (!canAct()) return false;
        const Piece& cur = *active_;
        if (cur.type == PieceType::O) return false;  // the shape does not change

        const Rotation to = clockwise ? rotatedCW(cur.rotation) : rotatedCCW(cur.rotation);
        const detail::KickTable& table =
            (cur.type == PieceType::I) ? detail::kIKicks : detail::kJlstzKicks;
        const detail::KickSet& kicks = table[detail::idx(cur.rotation)][clockwise ? 0 : 1];

        for (std::size_t i = 0; i < kicks.size(); ++i) {
            Piece candidate = cur;
            candidate.rotation = to;
            candidate.x += kicks[i].x;
            candidate.y -= kicks[i].y;  // convert y-up kick data to y-down
            if (board_.fits(candidate)) {
                active_ = candidate;
                lastMoveWasRotation_ = true;
                lastKickIndex_ = static_cast<int>(i);
                noteDescent();
                onSuccessfulManipulation();
                return true;
            }
        }
        return false;
    }

    // Move reset: a successful move or rotation restarts the lock timer if
    // the piece is in lock delay or grounded, up to maxLockResets times.
    void onSuccessfulManipulation() {
        if ((lockTimerMs_ > 0.0 || isGrounded()) && lockResets_ < config_.maxLockResets) {
            lockTimerMs_ = 0.0;
            ++lockResets_;
        }
    }

    // When the piece reaches a new lowest row, the reset budget refills.
    void noteDescent() {
        if (active_ && active_->y > lowestY_) {
            lowestY_ = active_->y;
            lockResets_ = 0;
            lockTimerMs_ = 0.0;
        }
    }

    void spawn(PieceType type) {
        Piece p;
        p.type = type;
        p.rotation = Rotation::Spawn;
        p.x = (config_.width - boxSize(type)) / 2;
        p.y = config_.hiddenRows - 2;

        if (!board_.fits(p)) {
            active_.reset();
            gameOver_ = GameOverReason::BlockOut;
            return;
        }
        active_ = p;
        // Guideline behaviour: drop one row right away if there is space.
        tryMove(0, 1);

        lowestY_ = active_->y;
        lockTimerMs_ = 0.0;
        lockResets_ = 0;
        gravityAccumMs_ = 0.0;
        lastMoveWasRotation_ = false;
        lastKickIndex_ = 0;
    }

    TSpin detectTSpin(const Piece& p) const noexcept {
        if (p.type != PieceType::T || !lastMoveWasRotation_) return TSpin::None;

        auto blocked = [&](int cx, int cy) { return board_.isBlocked(p.x + cx, p.y + cy); };
        const bool tl = blocked(0, 0);
        const bool tr = blocked(2, 0);
        const bool bl = blocked(0, 2);
        const bool br = blocked(2, 2);
        const int count = int{tl} + int{tr} + int{bl} + int{br};
        if (count < 3) return TSpin::None;

        bool frontA = false;
        bool frontB = false;
        switch (p.rotation) {
            case Rotation::Spawn:   frontA = tl; frontB = tr; break;
            case Rotation::Right:   frontA = tr; frontB = br; break;
            case Rotation::Reverse: frontA = bl; frontB = br; break;
            case Rotation::Left:    frontA = tl; frontB = bl; break;
        }
        if (frontA && frontB) return TSpin::Full;
        // The final SRS kick test always upgrades to a full T-spin.
        if (lastKickIndex_ == 4) return TSpin::Full;
        return TSpin::Mini;
    }

    static std::int64_t basePoints(int cleared, TSpin tspin) noexcept {
        static constexpr std::array<std::int64_t, 5> kNormal = {0, 100, 300, 500, 800};
        static constexpr std::array<std::int64_t, 4> kFull = {400, 800, 1200, 1600};
        static constexpr std::array<std::int64_t, 3> kMini = {100, 200, 400};
        const auto n = static_cast<std::size_t>(std::clamp(cleared, 0, 4));
        switch (tspin) {
            case TSpin::Full: return kFull[std::min<std::size_t>(n, 3)];
            case TSpin::Mini: return kMini[std::min<std::size_t>(n, 2)];
            case TSpin::None: break;
        }
        return kNormal[n];
    }

    static std::int64_t perfectClearPoints(int cleared, bool b2bTetris) noexcept {
        if (b2bTetris) return 3200;
        static constexpr std::array<std::int64_t, 4> kPc = {800, 1200, 1800, 2000};
        return kPc[static_cast<std::size_t>(std::clamp(cleared, 1, 4) - 1)];
    }

    void lockPiece() {
        const Piece p = *active_;
        const TSpin tspin = detectTSpin(p);

        bool entirelyHidden = true;
        for (const Point& c : p.cells()) {
            if (c.y >= config_.hiddenRows) {
                entirelyHidden = false;
                break;
            }
        }

        board_.place(p);
        active_.reset();
        ++piecesLocked_;

        const int cleared = board_.clearFullRows();
        const std::int64_t lvl = level_;

        LockResult result;
        result.piece = p.type;
        result.linesCleared = cleared;
        result.tspin = tspin;

        std::int64_t points = basePoints(cleared, tspin) * lvl;

        if (cleared > 0) {
            const bool difficult = (cleared >= 4) || (tspin != TSpin::None);
            const bool applyB2b = difficult && backToBack_;
            if (applyB2b) points = points * 3 / 2;
            result.backToBack = applyB2b;
            backToBack_ = difficult;

            ++combo_;
            if (combo_ > 0) points += 50 * static_cast<std::int64_t>(combo_) * lvl;

            if (board_.isEmpty()) {
                result.perfectClear = true;
                points += perfectClearPoints(cleared, applyB2b && cleared >= 4) * lvl;
            }

            lines_ += cleared;
            const int target = config_.startLevel + lines_ / config_.linesPerLevel;
            level_ = std::clamp(std::max(level_, target), 1, config_.maxLevel);
        } else {
            combo_ = -1;
        }

        result.combo = combo_;
        result.pointsAwarded = points;
        score_ += points;
        lastLock_ = result;
        canHold_ = true;

        if (entirelyHidden) {
            gameOver_ = GameOverReason::LockOut;
            return;
        }
        spawn(randomizer_.next());
    }

    GameConfig config_;
    Board board_;
    BagRandomizer randomizer_;

    std::optional<Piece> active_;
    std::optional<PieceType> held_;
    bool canHold_ = true;

    std::int64_t score_ = 0;
    int lines_ = 0;
    int level_ = 1;
    int combo_ = -1;
    bool backToBack_ = false;
    GameOverReason gameOver_ = GameOverReason::None;
    std::optional<LockResult> lastLock_;
    int piecesLocked_ = 0;

    double gravityAccumMs_ = 0.0;
    double lockTimerMs_ = 0.0;
    int lockResets_ = 0;
    int lowestY_ = 0;
    bool softDropActive_ = false;

    bool lastMoveWasRotation_ = false;
    int lastKickIndex_ = 0;
};

    do {                                                                            \
        if (!(cond)) {                                                              \

void testShapes() {
    // T piece in the Right orientation points to the right.
    const Minos& tRight = shapeOf(PieceType::T, Rotation::Right);
    bool hasRightNub = false;
    for (const Point& p : tRight) hasRightNub |= (p == Point{2, 1});
    BF_CHECK(hasRightNub);

    // The I piece in the Right orientation is the vertical bar in column 2.
    for (const Point& p : shapeOf(PieceType::I, Rotation::Right)) BF_CHECK(p.x == 2);

    // Four clockwise rotations return to the spawn shape.
    for (int t = 0; t < kPieceTypeCount; ++t) {
        const auto type = static_cast<PieceType>(t);
        Rotation r = Rotation::Spawn;
        for (int i = 0; i < 4; ++i) r = rotatedCW(r);
        BF_CHECK(r == Rotation::Spawn);
        const Minos& a = shapeOf(type, Rotation::Spawn);
        const Minos& b = shapeOf(type, r);
        for (std::size_t i = 0; i < 4; ++i) BF_CHECK(a[i] == b[i]);
    }
}

void testBoardClearing() {
    Board b(4, 4);
    for (int x = 0; x < 4; ++x) b.set(x, 3, 1);  // full bottom row
    b.set(0, 2, 2);                              // partial row
    for (int x = 0; x < 4; ++x) b.set(x, 1, 3);  // full row
    b.set(3, 0, 4);                              // top partial row

    BF_CHECK(b.fullRows().size() == 2);
    BF_CHECK(b.clearFullRows() == 2);
    BF_CHECK(b.at(0, 3) == 2);
    BF_CHECK(b.at(3, 2) == 4);
    BF_CHECK(b.isRowEmpty(0));
    BF_CHECK(b.isRowEmpty(1));
    BF_CHECK(b.fullRows().empty());
}

void testCollision() {
    Board b(10, 22);
    Piece p{PieceType::O, Rotation::Spawn, 3, 0};
    BF_CHECK(b.fits(p));
    BF_CHECK(!b.fits(p.translated(-5, 0)));  // through the left wall
    BF_CHECK(!b.fits(p.translated(0, 21)));  // below the floor
    b.set(4, 5, 1);
    BF_CHECK(!b.fits(p.translated(0, 5)));
}

void testHardDropAndClear() {
    GameConfig cfg;
    cfg.seed = 12345u;
    Game g(cfg);
    BF_CHECK(g.activePiece().has_value());
    BF_CHECK(g.preview().size() == 5);

    const auto ghost = g.ghostPiece();
    BF_CHECK(ghost.has_value());
    const int rows = g.hardDrop();
    BF_CHECK(rows > 0);
    BF_CHECK(g.piecesLocked() == 1);
    BF_CHECK(g.score() == 2 * rows);
}

void testHold() {
    GameConfig cfg;
    cfg.seed = 7u;
    Game g(cfg);
    const PieceType first = g.activePiece()->type;
    BF_CHECK(g.hold());
    BF_CHECK(g.heldPiece() == first);
    BF_CHECK(!g.hold());  // only one hold per piece
    g.hardDrop();
    BF_CHECK(g.canHold());
}

void testBagFairness() {
    BagRandomizer r(99u);
    for (int bag = 0; bag < 20; ++bag) {
        std::array<int, kPieceTypeCount> counts{};
        for (int i = 0; i < kPieceTypeCount; ++i) ++counts[static_cast<std::size_t>(r.next())];
        for (int c : counts) BF_CHECK(c == 1);
    }
}

void testGravityAndLock() {
    GameConfig cfg;
    cfg.seed = 1u;
    Game g(cfg);
    const int y0 = g.activePiece()->y;
    g.update(g.baseGravityIntervalMs() + 1.0);
    BF_CHECK(g.activePiece()->y == y0 + 1);

    // Run long enough for the first piece to fall and lock.
    for (int i = 0; i < 2000 && g.piecesLocked() == 0; ++i) g.update(16.0);
    BF_CHECK(g.piecesLocked() == 1);
}

void testRandomPlayTerminates() {
    GameConfig cfg;
    cfg.seed = 2024u;
    Game g(cfg);
    std::mt19937 rng(5u);
    std::uniform_int_distribution<int> action(0, 6);
    int steps = 0;
    while (!g.isGameOver() && steps < 200000) {
        switch (action(rng)) {
            case 0: g.moveLeft(); break;
            case 1: g.moveRight(); break;
            case 2: g.rotateCW(); break;
            case 3: g.rotateCCW(); break;
            case 4: g.softDrop(); break;
            case 5: g.hold(); break;
            default: g.hardDrop(); break;
        }
        g.update(16.0);
        ++steps;
    }
    BF_CHECK(g.isGameOver());
    BF_CHECK(g.gameOverReason() != GameOverReason::None);
}

    testCollision();
    testHardDropAndClear();
    testHold();
    testBagFairness();
    testGravityAndLock();
    testRandomPlayTerminates();

    if (g_failures == 0) {
        std::cout << "All blockfall self-tests passed.\n";
        return 0;
    }
    std::cerr << g_failures << " check(s) failed.\n";
    return 1;
}

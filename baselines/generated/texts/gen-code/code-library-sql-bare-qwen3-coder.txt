-- Library Lending System Schema
-- PostgreSQL 12+ compatible

-- Drop existing tables if they exist (for development purposes)
DROP TABLE IF EXISTS loan_items CASCADE;
DROP TABLE IF EXISTS loans CASCADE;
DROP TABLE IF EXISTS patrons CASCADE;
DROP TABLE IF EXISTS books CASCADE;
DROP TABLE IF EXISTS authors CASCADE;

-- Authors table
CREATE TABLE authors (
    author_id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    birth_date DATE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Books table
CREATE TABLE books (
    book_id SERIAL PRIMARY KEY,
    isbn VARCHAR(20) UNIQUE NOT NULL,
    title VARCHAR(500) NOT NULL,
    author_id INTEGER REFERENCES authors(author_id),
    publication_year INTEGER,
    genre VARCHAR(100),
    total_copies INTEGER DEFAULT 1 CHECK (total_copies >= 0),
    available_copies INTEGER DEFAULT 1 CHECK (available_copies >= 0 AND available_copies <= total_copies),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Patrons table
CREATE TABLE patrons (
    patron_id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE,
    phone VARCHAR(20),
    address TEXT,
    membership_date DATE DEFAULT CURRENT_DATE,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Loans table (header for loan transactions)
CREATE TABLE loans (
    loan_id SERIAL PRIMARY KEY,
    patron_id INTEGER NOT NULL REFERENCES patrons(patron_id),
    checkout_date DATE NOT NULL DEFAULT CURRENT_DATE,
    due_date DATE NOT NULL,
    return_date DATE,
    status VARCHAR(20) DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE', 'RETURNED', 'OVERDUE')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Loan items table (individual books in a loan)
CREATE TABLE loan_items (
    loan_item_id SERIAL PRIMARY KEY,
    loan_id INTEGER NOT NULL REFERENCES loans(loan_id) ON DELETE CASCADE,
    book_id INTEGER NOT NULL REFERENCES books(book_id),
    quantity INTEGER DEFAULT 1 CHECK (quantity > 0),
    returned_quantity INTEGER DEFAULT 0 CHECK (returned_quantity >= 0 AND returned_quantity <= quantity),
    status VARCHAR(20) DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE', 'RETURNED', 'PARTIALLY_RETURNED')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Indexes for performance
CREATE INDEX idx_books_isbn ON books(isbn);
CREATE INDEX idx_books_author ON books(author_id);
CREATE INDEX idx_loans_patron ON loans(patron_id);
CREATE INDEX idx_loans_status ON loans(status);
CREATE INDEX idx_loans_due_date ON loans(due_date);
CREATE INDEX idx_loan_items_loan ON loan_items(loan_id);
CREATE INDEX idx_loan_items_book ON loan_items(book_id);

-- Function to update book availability when copies change
CREATE OR REPLACE FUNCTION update_book_availability()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_update_book_timestamp
    BEFORE UPDATE ON books
    FOR EACH ROW
    EXECUTE FUNCTION update_book_availability();

-- Function to check out books
CREATE OR REPLACE FUNCTION checkout_book(
    p_patron_id INTEGER,
    p_book_id INTEGER,
    p_quantity INTEGER DEFAULT 1,
    p_loan_period_days INTEGER DEFAULT 14
)
RETURNS TABLE(loan_id INTEGER, message TEXT) AS $$
DECLARE
    v_loan_id INTEGER;
    v_available INTEGER;
    v_new_available INTEGER;
BEGIN
    -- Check if patron exists and is active
    IF NOT EXISTS (SELECT 1 FROM patrons WHERE patron_id = p_patron_id AND is_active = TRUE) THEN
        RETURN QUERY SELECT NULL::INTEGER, 'Patron not found or inactive'::TEXT;
        RETURN;
    END IF;

    -- Check if book exists
    IF NOT EXISTS (SELECT 1 FROM books WHERE book_id = p_book_id) THEN
        RETURN QUERY SELECT NULL::INTEGER, 'Book not found'::TEXT;
        RETURN;
    END IF;

    -- Check availability
    SELECT available_copies INTO v_available FROM books WHERE book_id = p_book_id;

    IF v_available < p_quantity THEN
        RETURN QUERY SELECT NULL::INTEGER, 'Not enough copies available. Available: ' || v_available::TEXT;
        RETURN;
    END IF;

    -- Create loan header
    INSERT INTO loans (patron_id, due_date)
    VALUES (p_patron_id, CURRENT_DATE + p_loan_period_days)
    RETURNING loans.loan_id INTO v_loan_id;

    -- Add loan item
    INSERT INTO loan_items (loan_id, book_id, quantity)
    VALUES (v_loan_id, p_book_id, p_quantity);

    -- Update book availability
    v_new_available := v_available - p_quantity;
    UPDATE books
    SET available_copies = v_new_available
    WHERE book_id = p_book_id;

    RETURN QUERY SELECT v_loan_id, 'Checkout successful'::TEXT;
END;
$$ LANGUAGE plpgsql;

-- Function to return books
CREATE OR REPLACE FUNCTION return_book(
    p_loan_id INTEGER,
    p_book_id INTEGER,
    p_quantity INTEGER DEFAULT 1
)
RETURNS TABLE(success BOOLEAN, message TEXT) AS $$
DECLARE
    v_loan_item_id INTEGER;
    v_current_quantity INTEGER;
    v_returned_quantity INTEGER;
    v_total_quantity INTEGER;
    v_new_returned_quantity INTEGER;
    v_book_copies INTEGER;
    v_item_status VARCHAR(20);
BEGIN
    -- Check if loan exists and is active
    IF NOT EXISTS (SELECT 1 FROM loans WHERE loan_id = p_loan_id AND status = 'ACTIVE') THEN
        RETURN QUERY SELECT FALSE, 'Loan not found or already completed';
        RETURN;
    END IF;

    -- Get loan item details
    SELECT li.loan_item_id, li.quantity, li.returned_quantity
    INTO v_loan_item_id, v_total_quantity, v_returned_quantity
    FROM loan_items li
    WHERE li.loan_id = p_loan_id AND li.book_id = p_book_id AND li.status = 'ACTIVE';

    IF NOT FOUND THEN
        RETURN QUERY SELECT FALSE, 'Book not found in this loan';
        RETURN;
    END IF;

    -- Check if trying to return more than borrowed
    IF (v_returned_quantity + p_quantity) > v_total_quantity THEN
        RETURN QUERY SELECT FALSE, 'Cannot return more copies than borrowed';
        RETURN;
    END IF;

    -- Update returned quantity
    v_new_returned_quantity := v_returned_quantity + p_quantity;

    IF v_new_returned_quantity = v_total_quantity THEN
        v_item_status := 'RETURNED';
    ELSE
        v_item_status := 'PARTIALLY_RETURNED';
    END IF;

    UPDATE loan_items
    SET returned_quantity = v_new_returned_quantity,
        status = v_item_status,
        updated_at = CURRENT_TIMESTAMP
    WHERE loan_item_id = v_loan_item_id;

    -- Update book availability
    SELECT available_copies INTO v_book_copies FROM books WHERE book_id = p_book_id;
    UPDATE books
    SET available_copies = v_book_copies + p_quantity
    WHERE book_id = p_book_id;

    -- Check if all items in loan are returned
    IF NOT EXISTS (
        SELECT 1 FROM loan_items
        WHERE loan_id = p_loan_id AND status IN ('ACTIVE', 'PARTIALLY_RETURNED')
    ) THEN
        UPDATE loans
        SET status = 'RETURNED',
            return_date = CURRENT_DATE,
            updated_at = CURRENT_TIMESTAMP
        WHERE loan_id = p_loan_id;
    END IF;

    RETURN QUERY SELECT TRUE, 'Return processed successfully';
END;
$$ LANGUAGE plpgsql;

-- Function to get overdue loans
CREATE OR REPLACE FUNCTION get_overdue_loans()
RETURNS TABLE(
    loan_id INTEGER,
    patron_id INTEGER,
    patron_name TEXT,
    book_title TEXT,
    checkout_date DATE,
    due_date DATE,
    days_overdue INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT
        l.loan_id,
        l.patron_id,
        CONCAT(p.first_name, ' ', p.last_name)::TEXT AS patron_name,
        b.title::TEXT AS book_title,
        l.checkout_date,
        l.due_date,
        (CURRENT_DATE - l.due_date)::INTEGER AS days_overdue
    FROM loans l
    JOIN patrons p ON l.patron_id = p.patron_id
    JOIN loan_items li ON l.loan_id = li.loan_id
    JOIN books b ON li.book_id = b.book_id
    WHERE l.status = 'ACTIVE'
      AND l.due_date < CURRENT_DATE
    ORDER BY l.due_date ASC;
END;
$$ LANGUAGE plpgsql;

-- Function to update overdue loan status
CREATE OR REPLACE FUNCTION update_overdue_status()
RETURNS VOID AS $$
BEGIN
    UPDATE loans
    SET status = 'OVERDUE',
        updated_at = CURRENT_TIMESTAMP
    WHERE status = 'ACTIVE'
      AND due_date < CURRENT_DATE
      AND return_date IS NULL;
END;
$$ LANGUAGE plpgsql;

-- View for current loans
CREATE OR REPLACE VIEW current_loans AS
SELECT
    l.loan_id,
    p.patron_id,
    CONCAT(p.first_name, ' ', p.last_name) AS patron_name,
    b.title AS book_title,
    l.checkout_date,
    l.due_date,
    li.quantity,
    (li.quantity - li.returned_quantity) AS remaining_quantity,
    CASE
        WHEN l.due_date < CURRENT_DATE THEN 'OVERDUE'
        ELSE 'ACTIVE'
    END AS loan_status
FROM loans l
JOIN patrons p ON l.patron_id = p.patron_id
JOIN loan_items li ON l.loan_id = li.loan_id
JOIN books b ON li.book_id = b.book_id
WHERE l.status IN ('ACTIVE', 'OVERDUE');

-- View for book availability
CREATE OR REPLACE VIEW book_availability AS
SELECT
    b.book_id,
    b.isbn,
    b.title,
    CONCAT(a.first_name, ' ', a.last_name) AS author_name,
    b.total_copies,
    b.available_copies,
    CASE
        WHEN b.available_copies > 0 THEN 'AVAILABLE'
        ELSE 'NOT_AVAILABLE'
    END AS availability_status
FROM books b
LEFT JOIN authors a ON b.author_id = a.author_id;

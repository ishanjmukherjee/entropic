#include <assert.h>
#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef void (*lru_free_fn)(void *value);

typedef struct lru_node {
    char *key;
    size_t key_len;
    uint64_t hash;
    void *value;
    struct lru_node *prev;
    struct lru_node *next;
    struct lru_node *hnext;
} lru_node_t;

typedef struct lru_cache {
    size_t capacity;
    size_t size;
    size_t bucket_count;
    lru_node_t **buckets;
    lru_node_t head; /* Sentinel: head.next is most recently used */
    lru_node_t tail; /* Sentinel: tail.prev is least recently used */
    lru_free_fn free_value;
} lru_cache_t;

static uint64_t fnv1a_hash(const char *data, size_t len) {
    uint64_t hash = 14695981039346656037ULL;
    for (size_t i = 0; i < len; i++) {
        hash ^= (uint8_t)data[i];
        hash *= 1099511628211ULL;
    }
    return hash;
}

static size_t next_power_of_two(size_t n) {
    if (n <= 16) {
        return 16;
    }
    n--;
    n |= n >> 1;
    n |= n >> 2;
    n |= n >> 4;
    n |= n >> 8;
    n |= n >> 16;
#if SIZE_MAX > 0xFFFFFFFFUL
    n |= n >> 32;
#endif
    return (n == SIZE_MAX) ? (SIZE_MAX >> 1) + 1 : n + 1;
}

static char *lru_strndup(const char *s, size_t len) {
    char *copy = (char *)malloc(len + 1);
    if (!copy) {
        return NULL;
    }
    memcpy(copy, s, len);
    copy[len] = '\0';
    return copy;
}

static inline void dll_remove(lru_node_t *node) {
    node->prev->next = node->next;
    node->next->prev = node->prev;
    node->prev = NULL;
    node->next = NULL;
}

static inline void dll_push_front(lru_cache_t *cache, lru_node_t *node) {
    node->next = cache->head.next;
    node->prev = &cache->head;
    cache->head.next->prev = node;
    cache->head.next = node;
}

static inline void dll_move_to_front(lru_cache_t *cache, lru_node_t *node) {
    if (cache->head.next == node) {
        return;
    }
    dll_remove(node);
    dll_push_front(cache, node);
}

static lru_node_t *ht_find(const lru_cache_t *cache, const char *key,
                           size_t key_len, uint64_t hash) {
    size_t idx = (size_t)(hash & (cache->bucket_count - 1));
    lru_node_t *curr = cache->buckets[idx];
    while (curr) {
        if (curr->hash == hash && curr->key_len == key_len &&
            memcmp(curr->key, key, key_len) == 0) {
            return curr;
        }
        curr = curr->hnext;
    }
    return NULL;
}

static void ht_insert(lru_cache_t *cache, lru_node_t *node) {
    size_t idx = (size_t)(node->hash & (cache->bucket_count - 1));
    node->hnext = cache->buckets[idx];
    cache->buckets[idx] = node;
}

static void ht_remove(lru_cache_t *cache, lru_node_t *node) {
    size_t idx = (size_t)(node->hash & (cache->bucket_count - 1));
    lru_node_t **pp = &cache->buckets[idx];
    while (*pp) {
        if (*pp == node) {
            *pp = node->hnext;
            node->hnext = NULL;
            return;
        }
        pp = &(*pp)->hnext;
    }
}

static void node_destroy(lru_cache_t *cache, lru_node_t *node) {
    if (cache->free_value && node->value) {
        cache->free_value(node->value);
    }
    free(node->key);
    free(node);
}

lru_cache_t *lru_create(size_t capacity, lru_free_fn free_value) {
    if (capacity == 0) {
        return NULL;
    }

    lru_cache_t *cache = (lru_cache_t *)calloc(1, sizeof(*cache));
    if (!cache) {
        return NULL;
    }

    size_t target_buckets =
        (capacity > SIZE_MAX / 2) ? SIZE_MAX : capacity * 2;
    size_t bucket_count = next_power_of_two(target_buckets);

    cache->buckets = (lru_node_t **)calloc(bucket_count, sizeof(lru_node_t *));
    if (!cache->buckets) {
        free(cache);
        return NULL;
    }

    cache->capacity = capacity;
    cache->size = 0;
    cache->bucket_count = bucket_count;
    cache->free_value = free_value;

    cache->head.prev = NULL;
    cache->head.next = &cache->tail;
    cache->tail.prev = &cache->head;
    cache->tail.next = NULL;

    return cache;
}

void lru_clear(lru_cache_t *cache) {
    if (!cache) {
        return;
    }

    lru_node_t *curr = cache->head.next;
    while (curr != &cache->tail) {
        lru_node_t *next = curr->next;
        node_destroy(cache, curr);
        curr = next;
    }

    memset(cache->buckets, 0, cache->bucket_count * sizeof(lru_node_t *));
    cache->head.next = &cache->tail;
    cache->tail.prev = &cache->head;
    cache->size = 0;
}

void lru_destroy(lru_cache_t *cache) {
    if (!cache) {
        return;
    }
    lru_clear(cache);
    free(cache->buckets);
    free(cache);
}

bool lru_get(lru_cache_t *cache, const char *key, void **out_value) {
    if (!cache || !key) {
        return false;
    }

    size_t key_len = strlen(key);
    uint64_t hash = fnv1a_hash(key, key_len);
    lru_node_t *node = ht_find(cache, key, key_len, hash);
    if (!node) {
        return false;
    }

    dll_move_to_front(cache, node);
    if (out_value) {
        *out_value = node->value;
    }
    return true;
}

int lru_put(lru_cache_t *cache, const char *key, void *value) {
    if (!cache || !key) {
        return -1;
    }

    size_t key_len = strlen(key);
    uint64_t hash = fnv1a_hash(key, key_len);
    lru_node_t *existing = ht_find(cache, key, key_len, hash);

    if (existing) {
        if (cache->free_value && existing->value && existing->value != value) {
            cache->free_value(existing->value);
        }
        existing->value = value;
        dll_move_to_front(cache, existing);
        return 0;
    }

    lru_node_t *node = (lru_node_t *)calloc(1, sizeof(*node));
    if (!node) {
        return -1;
    }

    node->key = lru_strndup(key, key_len);
    if (!node->key) {
        free(node);
        return -1;
    }

    node->key_len = key_len;
    node->hash = hash;
    node->value = value;

    if (cache->size == cache->capacity) {
        lru_node_t *victim = cache->tail.prev;
        dll_remove(victim);
        ht_remove(cache, victim);
        node_destroy(cache, victim);
        cache->size--;
    }

    ht_insert(cache, node);
    dll_push_front(cache, node);
    cache->size++;
    return 0;
}

bool lru_delete(lru_cache_t *cache, const char *key) {
    if (!cache || !key) {
        return false;
    }

    size_t key_len = strlen(key);
    uint64_t hash = fnv1a_hash(key, key_len);
    lru_node_t *node = ht_find(cache, key, key_len, hash);
    if (!node) {
        return false;
    }

    dll_remove(node);
    ht_remove(cache, node);
    node_destroy(cache, node);
    cache->size--;
    return true;
}

size_t lru_size(const lru_cache_t *cache) {
    return cache ? cache->size : 0;
}

size_t lru_capacity(const lru_cache_t *cache) {
    return cache ? cache->capacity : 0;
}

#ifdef LRU_CACHE_TEST
int main(void) {
    lru_cache_t *cache = lru_create(3, free);
    assert(cache != NULL);
    assert(lru_capacity(cache) == 3);
    assert(lru_size(cache) == 0);

    int *v1 = (int *)malloc(sizeof(int));
    int *v2 = (int *)malloc(sizeof(int));
    int *v3 = (int *)malloc(sizeof(int));
    int *v4 = (int *)malloc(sizeof(int));
    assert(v1 && v2 && v3 && v4);
    *v1 = 10;
    *v2 = 20;
    *v3 = 30;
    *v4 = 40;

    assert(lru_put(cache, "a", v1) == 0);
    assert(lru_put(cache, "b", v2) == 0);
    assert(lru_put(cache, "c", v3) == 0);
    assert(lru_size(cache) == 3);

    void *out = NULL;
    assert(lru_get(cache, "a", &out) && *(int *)out == 10);

    /* "b" is now LRU and should be evicted */
    assert(lru_put(cache, "d", v4) == 0);
    assert(lru_size(cache) == 3);
    assert(!lru_get(cache, "b", &out));
    assert(lru_get(cache, "a", &out) && *(int *)out == 10);
    assert(lru_get(cache, "c", &out) && *(int *)out == 30);
    assert(lru_get(cache, "d", &out) && *(int *)out == 40);

    /* Update existing key */
    int *v3_new = (int *)malloc(sizeof(int));
    assert(v3_new);
    *v3_new = 300;
    assert(lru_put(cache, "c", v3_new) == 0);
    assert(lru_get(cache, "c", &out) && *(int *)out == 300);

    /* Delete key */
    assert(lru_delete(cache, "a"));
    assert(!lru_delete(cache, "a"));
    assert(lru_size(cache) == 2);

    lru_clear(cache);
    assert(lru_size(cache) == 0);

    lru_destroy(cache);
    printf("All LRU cache tests passed.\n");
    return 0;
}
#endif

//
//  ArticleList.swift
//
//  A paginated, pull-to-refreshable list of articles backed by a JSON API.
//
//  Expected API shape:
//
//  GET {baseURL}/articles?page=1&per_page=20
//  {
//    "data": [
//      {
//        "id": "abc123",
//        "title": "Hello",
//        "summary": "…",
//        "author": "Jane Doe",
//        "published_at": "2025-01-01T12:00:00.000Z",
//        "image_url": "https://…",
//        "url": "https://…"
//      }
//    ],
//    "meta": { "page": 1, "per_page": 20, "total_pages": 5 }
//  }
//
//  Requires iOS 16+ / macOS 13+.
//

import SwiftUI

// MARK: - Models

struct Article: Identifiable, Hashable, Sendable, Decodable {
    let id: String
    let title: String
    let summary: String?
    let author: String?
    let publishedAt: Date?
    let imageURL: URL?
    let url: URL?

    private enum CodingKeys: String, CodingKey {
        case id, title, summary, author, url
        case publishedAt = "published_at"
        case imageURL = "image_url"
    }

    init(
        id: String,
        title: String,
        summary: String? = nil,
        author: String? = nil,
        publishedAt: Date? = nil,
        imageURL: URL? = nil,
        url: URL? = nil
    ) {
        self.id = id
        self.title = title
        self.summary = summary
        self.author = author
        self.publishedAt = publishedAt
        self.imageURL = imageURL
        self.url = url
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        // Accept both string and integer identifiers.
        if let stringID = try? container.decode(String.self, forKey: .id) {
            id = stringID
        } else {
            id = String(try container.decode(Int.self, forKey: .id))
        }
        title = try container.decode(String.self, forKey: .title)
        summary = try container.decodeIfPresent(String.self, forKey: .summary)
        author = try container.decodeIfPresent(String.self, forKey: .author)
        publishedAt = try container.decodeIfPresent(Date.self, forKey: .publishedAt)
        // Tolerate malformed or empty URLs rather than failing the whole page.
        imageURL = (try? container.decodeIfPresent(String.self, forKey: .imageURL)).flatMap { $0 }.flatMap(URL.init(string:))
        url = (try? container.decodeIfPresent(String.self, forKey: .url)).flatMap { $0 }.flatMap(URL.init(string:))
    }
}

struct ArticlePage: Sendable, Decodable {
    struct Meta: Sendable, Decodable {
        let page: Int
        let perPage: Int
        let totalPages: Int

        private enum CodingKeys: String, CodingKey {
            case page
            case perPage = "per_page"
            case totalPages = "total_pages"
        }
    }

    let data: [Article]
    let meta: Meta

    var hasMorePages: Bool { meta.page < meta.totalPages }
}

// MARK: - Networking

protocol ArticleFetching: Sendable {
    func fetchArticles(page: Int, pageSize: Int) async throws -> ArticlePage
}

enum ArticleAPIError: LocalizedError {
    case invalidURL
    case invalidResponse
    case httpStatus(Int)
    case decoding(Error)

    var errorDescription: String? {
        switch self {
        case .invalidURL:
            return "The request URL was invalid."
        case .invalidResponse:
            return "The server returned an invalid response."
        case .httpStatus(let code):
            switch code {
            case 401, 403: return "You're not authorized to view these articles."
            case 404: return "Articles could not be found."
            case 429: return "Too many requests. Please try again shortly."
            case 500...599: return "The server is having trouble right now. Please try again later."
            default: return "The request failed (HTTP \(code))."
            }
        case .decoding:
            return "The articles couldn't be read."
        }
    }
}

struct ArticleAPIClient: ArticleFetching {
    let baseURL: URL
    private let session: URLSession
    private let decoder: JSONDecoder

    init(baseURL: URL, session: URLSession = .shared) {
        self.baseURL = baseURL
        self.session = session
        self.decoder = Self.makeDecoder()
    }

    func fetchArticles(page: Int, pageSize: Int) async throws -> ArticlePage {
        var components = URLComponents(
            url: baseURL.appendingPathComponent("articles"),
            resolvingAgainstBaseURL: false
        )
        components?.queryItems = [
            URLQueryItem(name: "page", value: String(page)),
            URLQueryItem(name: "per_page", value: String(pageSize)),
        ]
        guard let url = components?.url else { throw ArticleAPIError.invalidURL }

        var request = URLRequest(url: url, timeoutInterval: 30)
        request.httpMethod = "GET"
        request.setValue("application/json", forHTTPHeaderField: "Accept")

        let (data, response) = try await session.data(for: request)

        guard let http = response as? HTTPURLResponse else {
            throw ArticleAPIError.invalidResponse
        }
        guard (200..<300).contains(http.statusCode) else {
            throw ArticleAPIError.httpStatus(http.statusCode)
        }

        do {
            return try decoder.decode(ArticlePage.self, from: data)
        } catch {
            throw ArticleAPIError.decoding(error)
        }
    }

    private static func makeDecoder() -> JSONDecoder {
        let decoder = JSONDecoder()
        let withFraction = Date.ISO8601FormatStyle(includingFractionalSeconds: true)
        let withoutFraction = Date.ISO8601FormatStyle()
        decoder.dateDecodingStrategy = .custom { decoder in
            let container = try decoder.singleValueContainer()
            let string = try container.decode(String.self)
            if let date = try? withFraction.parse(string) { return date }
            if let date = try? withoutFraction.parse(string) { return date }
            throw DecodingError.dataCorruptedError(
                in: container,
                debugDescription: "Invalid ISO 8601 date: \(string)"
            )
        }
        return decoder
    }
}

// MARK: - View Model

@MainActor
final class ArticleListViewModel: ObservableObject {
    enum Phase: Equatable {
        case idle
        case loading
        case loaded
        case empty
        case failed(message: String)
    }

    @Published private(set) var articles: [Article] = []
    @Published private(set) var phase: Phase = .idle
    @Published private(set) var isLoadingNextPage = false
    @Published private(set) var nextPageErrorMessage: String?
    @Published private(set) var hasMorePages = true
    @Published var refreshErrorMessage: String?

    private let service: ArticleFetching
    private let pageSize: Int
    private let prefetchThreshold: Int

    private var nextPage = 1
    private var seenIDs = Set<Article.ID>()
    private var nextPageTask: Task<Void, Never>?
    /// Incremented whenever the list is reset, so stale responses can be discarded.
    private var generation = 0

    init(service: ArticleFetching, pageSize: Int = 20, prefetchThreshold: Int = 5) {
        precondition(pageSize > 0, "pageSize must be positive")
        self.service = service
        self.pageSize = pageSize
        self.prefetchThreshold = max(0, prefetchThreshold)
    }

    deinit {
        nextPageTask?.cancel()
    }

    // MARK: Public API

    /// Loads the first page once. Safe to call repeatedly (e.g. from `.task`).
    func loadInitialIfNeeded() async {
        guard phase == .idle else { return }
        await loadFirstPage(isRefresh: false)
    }

    /// Reloads from the first page after a full-screen failure.
    func retry() async {
        await loadFirstPage(isRefresh: false)
    }

    /// Pull-to-refresh. Keeps existing content visible while loading.
    func refresh() async {
        await loadFirstPage(isRefresh: true)
    }

    /// Call when a row appears; triggers the next page when nearing the end.
    func loadNextPageIfNeeded(currentItem: Article) {
        guard shouldLoadNextPage else { return }
        guard let index = articles.lastIndex(where: { $0.id == currentItem.id }) else { return }
        guard index >= articles.count - 1 - prefetchThreshold else { return }
        loadNextPage()
    }

    /// Retries a failed next-page request.
    func retryNextPage() {
        nextPageErrorMessage = nil
        guard shouldLoadNextPage else { return }
        loadNextPage()
    }

    // MARK: Private

    private var shouldLoadNextPage: Bool {
        phase == .loaded && hasMorePages && !isLoadingNextPage && nextPageErrorMessage == nil
    }

    private func loadFirstPage(isRefresh: Bool) async {
        nextPageTask?.cancel()
        nextPageTask = nil
        isLoadingNextPage = false
        nextPageErrorMessage = nil

        generation += 1
        let currentGeneration = generation

        let hasExistingContent = !articles.isEmpty
        if !isRefresh || !hasExistingContent {
            phase = .loading
        }

        do {
            let page = try await service.fetchArticles(page: 1, pageSize: pageSize)
            guard currentGeneration == generation else { return }
            apply(page, reset: true)
        } catch {
            guard currentGeneration == generation else { return }

            if Self.isCancellation(error) {
                // Restore a sensible phase if the load was cancelled mid-flight.
                if phase == .loading {
                    phase = hasExistingContent ? .loaded : .idle
                }
                return
            }

            let message = Self.message(for: error)
            if isRefresh && hasExistingContent {
                phase = .loaded
                refreshErrorMessage = message
            } else {
                articles = []
                seenIDs = []
                phase = .failed(message: message)
            }
        }
    }

    private func loadNextPage() {
        isLoadingNextPage = true
        nextPageErrorMessage = nil

        let currentGeneration = generation
        let pageToLoad = nextPage
        let service = self.service
        let pageSize = self.pageSize

        nextPageTask = Task { [weak self] in
            do {
                let page = try await service.fetchArticles(page: pageToLoad, pageSize: pageSize)
                guard let self, currentGeneration == self.generation else { return }
                self.isLoadingNextPage = false
                self.apply(page, reset: false)
            } catch {
                guard let self, currentGeneration == self.generation else { return }
                self.isLoadingNextPage = false
                guard !Self.isCancellation(error) else { return }
                self.nextPageErrorMessage = Self.message(for: error)
            }
        }
    }

    private func apply(_ page: ArticlePage, reset: Bool) {
        if reset {
            articles = []
            seenIDs = []
        }
        // Guard against duplicates caused by items shifting between pages.
        let newItems = page.data.filter { seenIDs.insert($0.id).inserted }
        articles.append(contentsOf: newItems)

        nextPage = page.meta.page + 1
        // Stop paginating if the server reports more pages but returns nothing.
        hasMorePages = page.hasMorePages && !page.data.isEmpty
        phase = articles.isEmpty ? .empty : .loaded
    }

    private static func isCancellation(_ error: Error) -> Bool {
        if error is CancellationError { return true }
        if let urlError = error as? URLError, urlError.code == .cancelled { return true }
        return false
    }

    private static func message(for error: Error) -> String {
        if let urlError = error as? URLError {
            switch urlError.code {
            case .notConnectedToInternet, .networkConnectionLost, .dataNotAllowed:
                return "You appear to be offline. Check your connection and try again."
            case .timedOut:
                return "The request timed out. Please try again."
            default:
                break
            }
        }
        return (error as? LocalizedError)?.errorDescription ?? error.localizedDescription
    }
}

// MARK: - Views

struct ArticleListView: View {
    @StateObject private var viewModel: ArticleListViewModel

    init(service: ArticleFetching, pageSize: Int = 20) {
        _viewModel = StateObject(
            wrappedValue: ArticleListViewModel(service: service, pageSize: pageSize)
        )
    }

    init(viewModel: @autoclosure @escaping () -> ArticleListViewModel) {
        _viewModel = StateObject(wrappedValue: viewModel())
    }

    var body: some View {
        NavigationStack {
            content
                .navigationTitle("Articles")
                .navigationDestination(for: Article.self) { article in
                    ArticleDetailView(article: article)
                }
        }
        .task { await viewModel.loadInitialIfNeeded() }
        .alert(
            "Couldn't Refresh",
            isPresented: Binding(
                get: { viewModel.refreshErrorMessage != nil },
                set: { if !$0 { viewModel.refreshErrorMessage = nil } }
            ),
            actions: { Button("OK", role: .cancel) {} },
            message: { Text(viewModel.refreshErrorMessage ?? "") }
        )
    }

    @ViewBuilder
    private var content: some View {
        switch viewModel.phase {
        case .idle, .loading:
            ProgressView("Loading articles…")
                .frame(maxWidth: .infinity, maxHeight: .infinity)

        case .empty:
            StatusMessageView(
                systemImage: "doc.text.magnifyingglass",
                title: "No Articles",
                message: "There's nothing here yet. Pull to refresh or check back later.",
                actionTitle: "Refresh",
                action: { await viewModel.refresh() }
            )

        case .failed(let message):
            StatusMessageView(
                systemImage: "exclamationmark.triangle",
                title: "Something Went Wrong",
                message: message,
                actionTitle: "Try Again",
                action: { await viewModel.retry() }
            )

        case .loaded:
            articleList
        }
    }

    private var articleList: some View {
        List {
            ForEach(viewModel.articles) { article in
                NavigationLink(value: article) {
                    ArticleRow(article: article)
                }
                .onAppear { viewModel.loadNextPageIfNeeded(currentItem: article) }
            }

            PaginationFooter(
                isLoading: viewModel.isLoadingNextPage,
                errorMessage: viewModel.nextPageErrorMessage,
                hasMorePages: viewModel.hasMorePages,
                onRetry: viewModel.retryNextPage
            )
        }
        .listStyle(.plain)
        .refreshable { await viewModel.refresh() }
    }
}

private struct ArticleRow: View {
    let article: Article

    var body: some View {
        HStack(alignment: .top, spacing: 12) {
            ArticleThumbnail(url: article.imageURL)

            VStack(alignment: .leading, spacing: 4) {
                Text(article.title)
                    .font(.headline)
                    .lineLimit(2)

                if let summary = article.summary, !summary.isEmpty {
                    Text(summary)
                        .font(.subheadline)
                        .foregroundStyle(.secondary)
                        .lineLimit(2)
                }

                metadata
            }
        }
        .padding(.vertical, 4)
        .accessibilityElement(children: .combine)
    }

    @ViewBuilder
    private var metadata: some View {
        let parts: [String] = [
            article.author,
            article.publishedAt.map { $0.formatted(.relative(presentation: .named)) },
        ].compactMap { $0 }.filter { !$0.isEmpty }

        if !parts.isEmpty {
            Text(parts.joined(separator: " · "))
                .font(.caption)
                .foregroundStyle(.tertiary)
                .lineLimit(1)
        }
    }
}

private struct ArticleThumbnail: View {
    let url: URL?
    private let size: CGFloat = 64

    var body: some View {
        Group {
            if let url {
                AsyncImage(url: url, transaction: Transaction(animation: .easeInOut)) { phase in
                    switch phase {
                    case .success(let image):
                        image.resizable().scaledToFill()
                    case .failure:
                        placeholder
                    case .empty:
                        ProgressView()
                    @unknown default:
                        placeholder
                    }
                }
            } else {
                placeholder
            }
        }
        .frame(width: size, height: size)
        .background(Color.secondary.opacity(0.1))
        .clipShape(RoundedRectangle(cornerRadius: 8, style: .continuous))
        .accessibilityHidden(true)
    }

    private var placeholder: some View {
        Image(systemName: "photo")
            .foregroundStyle(.secondary)
    }
}

private struct PaginationFooter: View {
    let isLoading: Bool
    let errorMessage: String?
    let hasMorePages: Bool
    let onRetry: () -> Void

    var body: some View {
        Group {
            if isLoading {
                ProgressView()
                    .accessibilityLabel("Loading more articles")
            } else if let errorMessage {
                VStack(spacing: 8) {
                    Text(errorMessage)
                        .font(.footnote)
                        .foregroundStyle(.secondary)
                        .multilineTextAlignment(.center)
                    Button("Retry", action: onRetry)
                        .buttonStyle(.bordered)
                }
            } else if !hasMorePages {
                Text("You're all caught up.")
                    .font(.footnote)
                    .foregroundStyle(.secondary)
            }
        }
        .frame(maxWidth: .infinity)
        .padding(.vertical, 12)
        .listRowSeparator(.hidden)
    }
}

private struct StatusMessageView: View {
    let systemImage: String
    let title: String
    let message: String
    let actionTitle: String
    let action: () async -> Void

    @State private var isPerformingAction = false

    var body: some View {
        VStack(spacing: 16) {
            Image(systemName: systemImage)
                .font(.system(size: 44))
                .foregroundStyle(.secondary)
                .accessibilityHidden(true)

            Text(title)
                .font(.title3.weight(.semibold))

            Text(message)
                .font(.body)
                .foregroundStyle(.secondary)
                .multilineTextAlignment(.center)

            Button {
                Task {
                    isPerformingAction = true
                    await action()
                    isPerformingAction = false
                }
            } label: {
                if isPerformingAction {
                    ProgressView()
                } else {
                    Text(actionTitle)
                }
            }
            .buttonStyle(.borderedProminent)
            .disabled(isPerformingAction)
        }
        .padding(32)
        .frame(maxWidth: .infinity, maxHeight: .infinity)
    }
}

private struct ArticleDetailView: View {
    let article: Article

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 16) {
                if let imageURL = article.imageURL {
                    AsyncImage(url: imageURL) { image in
                        image.resizable().scaledToFit()
                    } placeholder: {
                        Rectangle()
                            .fill(Color.secondary.opacity(0.1))
                            .aspectRatio(16 / 9, contentMode: .fit)
                    }
                    .clipShape(RoundedRectangle(cornerRadius: 12, style: .continuous))
                    .accessibilityHidden(true)
                }

                Text(article.title)
                    .font(.title2.weight(.bold))

                if article.author != nil || article.publishedAt != nil {
                    HStack(spacing: 8) {
                        if let author = article.author { Text(author) }
                        if let date = article.publishedAt {
                            Text(date.formatted(date: .abbreviated, time: .shortened))
                        }
                    }
                    .font(.subheadline)
                    .foregroundStyle(.secondary)
                }

                if let summary = article.summary {
                    Text(summary).font(.body)
                }

                if let url = article.url {
                    Link(destination: url) {
                        Label("Read Full Article", systemImage: "safari")
                    }
                    .buttonStyle(.bordered)
                }
            }
            .padding()
            .frame(maxWidth: .infinity, alignment: .leading)
        }
        .navigationTitle("Article")
        #if os(iOS)
        .navigationBarTitleDisplayMode(.inline)
        #endif
    }
}

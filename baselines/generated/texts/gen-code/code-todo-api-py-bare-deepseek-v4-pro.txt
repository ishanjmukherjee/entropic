#!/usr/bin/env python3
"""
A production-quality JSON HTTP API for managing a to-do list.
Uses only Python's standard library.

Endpoints:
  GET    /tasks           - List all tasks
  GET    /tasks/<id>      - Get a single task
  POST   /tasks           - Create a task (body: {"title": "...", "notes": "...", "due": "..."})
  PUT    /tasks/<id>      - Update a task (body: same as create, fields optional)
  DELETE /tasks/<id>      - Delete a task
"""

import json
import os
import signal
import sqlite3
import sys
import uuid
from datetime import datetime, timezone
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, unquote
from typing import Optional, Dict, Any, List

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
DB_PATH = os.environ.get("TODO_DB_PATH", os.path.join(os.path.dirname(os.path.abspath(__file__)), "todo.db"))
HOST = os.environ.get("TODO_HOST", "0.0.0.0")
PORT = int(os.environ.get("TODO_PORT", "8080"))


# ---------------------------------------------------------------------------
# Database helpers
# ---------------------------------------------------------------------------
def get_db() -> sqlite3.Connection:
    """Return a connection to the SQLite database with WAL mode and foreign keys enabled."""
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    return conn


def init_db() -> None:
    """Create tables if they don't exist."""
    with get_db() as conn:
        conn.execute("""
            CREATE TABLE IF NOT EXISTS tasks (
                id          TEXT PRIMARY KEY,
                title       TEXT NOT NULL,
                notes       TEXT NOT NULL DEFAULT '',
                due         TEXT DEFAULT NULL,
                completed   INTEGER NOT NULL DEFAULT 0,
                created_at  TEXT NOT NULL,
                updated_at  TEXT NOT NULL
            )
        """)
        # Migration for any existing schema adjustments
        conn.execute("""
            CREATE TABLE IF NOT EXISTS schema_version (
                version INTEGER PRIMARY KEY
            )
        """)
        cur = conn.execute("SELECT MAX(version) FROM schema_version")
        version = cur.fetchone()[0]
        if version is None:
            conn.execute("INSERT INTO schema_version (version) VALUES (1)")
            conn.commit()


# ---------------------------------------------------------------------------
# Data transfer utilities
# ---------------------------------------------------------------------------
def row_to_dict(row: sqlite3.Row) -> Dict[str, Any]:
    """Convert a database row to a JSON-serializable dictionary."""
    return {
        "id": row["id"],
        "title": row["title"],
        "notes": row["notes"],
        "due": row["due"],
        "completed": bool(row["completed"]),
        "created_at": row["created_at"],
        "updated_at": row["updated_at"],
    }


def now_iso() -> str:
    """Return current UTC datetime as an ISO 8601 string."""
    return datetime.now(timezone.utc).isoformat()


# ---------------------------------------------------------------------------
# Task CRUD
# ---------------------------------------------------------------------------
def list_tasks() -> List[Dict[str, Any]]:
    """Return all tasks ordered by creation time descending."""
    with get_db() as conn:
        rows = conn.execute("SELECT * FROM tasks ORDER BY created_at DESC").fetchall()
        return [row_to_dict(r) for r in rows]


def get_task(task_id: str) -> Optional[Dict[str, Any]]:
    """Return a single task by ID, or None."""
    with get_db() as conn:
        row = conn.execute("SELECT * FROM tasks WHERE id = ?", (task_id,)).fetchone()
        return row_to_dict(row) if row else None


def create_task(data: Dict[str, Any]) -> Dict[str, Any]:
    """Create a new task. Validates title presence."""
    title = (data.get("title") or "").strip()
    if not title:
        raise ValueError("title is required and must not be empty")

    task_id = uuid.uuid4().hex
    notes = data.get("notes", "")
    due = data.get("due")
    completed = bool(data.get("completed", False))
    ts = now_iso()

    with get_db() as conn:
        conn.execute(
            """INSERT INTO tasks (id, title, notes, due, completed, created_at, updated_at)
               VALUES (?, ?, ?, ?, ?, ?, ?)""",
            (task_id, title, notes, due, int(completed), ts, ts),
        )
        conn.commit()
    return get_task(task_id)  # type: ignore[return-value]


def update_task(task_id: str, data: Dict[str, Any]) -> Optional[Dict[str, Any]]:
    """Update an existing task. Only provided fields are changed."""
    existing = get_task(task_id)
    if existing is None:
        return None

    title = data.get("title", existing["title"])
    notes = data.get("notes", existing["notes"])
    due = data.get("due", existing["due"])
    if "completed" in data:
        completed = bool(data["completed"])
    else:
        completed = existing["completed"]

    if not title or not title.strip():
        raise ValueError("title must not be empty")

    ts = now_iso()
    with get_db() as conn:
        conn.execute(
            """UPDATE tasks
               SET title = ?, notes = ?, due = ?, completed = ?, updated_at = ?
               WHERE id = ?""",
            (title.strip(), notes, due, int(completed), ts, task_id),
        )
        conn.commit()
    return get_task(task_id)


def delete_task(task_id: str) -> bool:
    """Delete a task. Returns True if a row was deleted."""
    with get_db() as conn:
        cur = conn.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
        conn.commit()
        return cur.rowcount > 0


# ---------------------------------------------------------------------------
# HTTP handler
# ---------------------------------------------------------------------------
class TodoHandler(BaseHTTPRequestHandler):
    """Request handler for the to-do API."""

    server_version = "TodoAPI/1.0"

    def _send_json(self, code: int, payload: Any) -> None:
        """Send a JSON response with the given status code."""
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _read_body(self) -> Dict[str, Any]:
        """Read and parse the JSON request body."""
        length = int(self.headers.get("Content-Length", 0))
        if length == 0:
            return {}
        raw = self.rfile.read(length)
        try:
            data = json.loads(raw)
        except json.JSONDecodeError:
            raise ValueError("Invalid JSON in request body")
        if not isinstance(data, dict):
            raise ValueError("Request body must be a JSON object")
        return data

    def _extract_id(self) -> str:
        """Extract a task ID from the URL path. Returns empty string for /tasks."""
        parsed = urlparse(self.path)
        path = unquote(parsed.path).rstrip("/")
        parts = [p for p in path.split("/") if p]
        # Expected forms: ['tasks'] or ['tasks', '<id>']
        if len(parts) >= 2 and parts[0] == "tasks":
            return parts[1]
        return ""

    # ------------------------------------------------------------------
    # Route dispatching
    # ------------------------------------------------------------------
    def do_GET(self) -> None:
        parsed = urlparse(self.path)
        path = unquote(parsed.path).rstrip("/")
        parts = [p for p in path.split("/") if p]

        if len(parts) == 1 and parts[0] == "tasks":
            tasks = list_tasks()
            self._send_json(200, tasks)
        elif len(parts) == 2 and parts[0] == "tasks":
            task_id = parts[1]
            task = get_task(task_id)
            if task is None:
                self._send_json(404, {"error": "Task not found"})
            else:
                self._send_json(200, task)
        else:
            self._send_json(404, {"error": "Not found"})

    def do_POST(self) -> None:
        parsed = urlparse(self.path)
        path = unquote(parsed.path).rstrip("/")

        if path != "/tasks":
            self._send_json(404, {"error": "Not found"})
            return

        try:
            data = self._read_body()
            task = create_task(data)
            self._send_json(201, task)
        except ValueError as exc:
            self._send_json(400, {"error": str(exc)})
        except Exception:
            self._send_json(500, {"error": "Internal server error"})

    def do_PUT(self) -> None:
        parsed = urlparse(self.path)
        path = unquote(parsed.path).rstrip("/")
        parts = [p for p in path.split("/") if p]

        if not (len(parts) == 2 and parts[0] == "tasks"):
            self._send_json(404, {"error": "Not found"})
            return

        task_id = parts[1]
        try:
            data = self._read_body()
            updated = update_task(task_id, data)
            if updated is None:
                self._send_json(404, {"error": "Task not found"})
            else:
                self._send_json(200, updated)
        except ValueError as exc:
            self._send_json(400, {"error": str(exc)})
        except Exception:
            self._send_json(500, {"error": "Internal server error"})

    def do_DELETE(self) -> None:
        parsed = urlparse(self.path)
        path = unquote(parsed.path).rstrip("/")
        parts = [p for p in path.split("/") if p]

        if not (len(parts) == 2 and parts[0] == "tasks"):
            self._send_json(404, {"error": "Not found"})
            return

        task_id = parts[1]
        deleted = delete_task(task_id)
        if not deleted:
            self._send_json(404, {"error": "Task not found"})
        else:
            self._send_json(200, {"message": "Task deleted"})

    # Override to avoid sending full exception tracebacks to clients
    def log_message(self, format: str, *args: Any) -> None:
        """Log to stderr in a simplified format."""
        sys.stderr.write(f"{self.address_string()} - {format % args}\n")


# ---------------------------------------------------------------------------
# Server runner
# ---------------------------------------------------------------------------
def main() -> None:
    """Initialise database and start the HTTP server."""
    init_db()
    server = HTTPServer((HOST, PORT), TodoHandler)

    # Graceful shutdown on SIGINT / SIGTERM
    def shutdown(signum: int, frame: Any) -> None:
        print("\nShutting down server...", file=sys.stderr)
        server.shutdown()

    signal.signal(signal.SIGINT, shutdown)
    signal.signal(signal.SIGTERM, shutdown)

    print(f"Todo API listening on http://{HOST}:{PORT}", file=sys.stderr)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()
        print("Server stopped.", file=sys.stderr)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
A small JSON HTTP API for a to-do list, built only on the Python standard library.

Endpoints
---------
GET    /health                  Liveness check.
GET    /todos                   List todos. Optional query: ?completed=true|false
POST   /todos                   Create a todo. Body: {"title": str, "completed"?: bool}
GET    /todos/{id}              Fetch a single todo.
PUT    /todos/{id}              Replace a todo. Body: {"title": str, "completed": bool}
PATCH  /todos/{id}              Partially update a todo. Body: {"title"?: str, "completed"?: bool}
DELETE /todos/{id}              Delete a todo.

Usage
-----
    python todo_api.py --host 127.0.0.1 --port 8000 --data-file todos.json

If --data-file is omitted, todos are kept in memory only.
"""

from __future__ import annotations

import argparse
import json
import logging
import os
import re
import signal
import sys
import tempfile
import threading
import uuid
from dataclasses import asdict, dataclass
from datetime import datetime, timezone
from http import HTTPStatus
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from typing import Any, Callable, Dict, List, Optional, Tuple
from urllib.parse import parse_qs, urlsplit

LOG = logging.getLogger("todo_api")

MAX_BODY_BYTES = 64 * 1024
MAX_TITLE_LENGTH = 500
API_VERSION = "1.0.0"


# --------------------------------------------------------------------------- #
# Errors
# --------------------------------------------------------------------------- #


class APIError(Exception):
    """An error that maps directly onto an HTTP error response."""

    def __init__(
        self,
        status: HTTPStatus,
        message: str,
        details: Optional[Dict[str, Any]] = None,
        headers: Optional[Dict[str, str]] = None,
    ) -> None:
        super().__init__(message)
        self.status = status
        self.message = message
        self.details = details or {}
        self.headers = headers or {}

    def to_dict(self) -> Dict[str, Any]:
        body: Dict[str, Any] = {
            "error": {
                "status": int(self.status),
                "code": self.status.phrase,
                "message": self.message,
            }
        }
        if self.details:
            body["error"]["details"] = self.details
        return body


# --------------------------------------------------------------------------- #
# Model and validation
# --------------------------------------------------------------------------- #


def utc_now_iso() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="milliseconds").replace("+00:00", "Z")


@dataclass
class Todo:
    id: str
    title: str
    completed: bool
    created_at: str
    updated_at: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "Todo":
        return cls(
            id=str(data["id"]),
            title=str(data["title"]),
            completed=bool(data["completed"]),
            created_at=str(data["created_at"]),
            updated_at=str(data["updated_at"]),
        )


ALLOWED_FIELDS = {"title", "completed"}


def _validate_title(value: Any) -> str:
    if not isinstance(value, str):
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY, "Field 'title' must be a string.",
                       {"field": "title"})
    title = value.strip()
    if not title:
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY, "Field 'title' must not be empty.",
                       {"field": "title"})
    if len(title) > MAX_TITLE_LENGTH:
        raise APIError(
            HTTPStatus.UNPROCESSABLE_ENTITY,
            f"Field 'title' must be at most {MAX_TITLE_LENGTH} characters.",
            {"field": "title", "max_length": MAX_TITLE_LENGTH},
        )
    return title


def _validate_completed(value: Any) -> bool:
    # bool is a subclass of int; reject ints explicitly by checking type identity.
    if type(value) is not bool:
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY, "Field 'completed' must be a boolean.",
                       {"field": "completed"})
    return value


def validate_payload(payload: Any, *, required: Tuple[str, ...]) -> Dict[str, Any]:
    """Validate a JSON object payload, returning only cleaned, known fields."""
    if not isinstance(payload, dict):
        raise APIError(HTTPStatus.BAD_REQUEST, "Request body must be a JSON object.")

    unknown = sorted(set(payload) - ALLOWED_FIELDS)
    if unknown:
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY, "Unknown field(s) in request body.",
                       {"unknown_fields": unknown, "allowed_fields": sorted(ALLOWED_FIELDS)})

    missing = [name for name in required if name not in payload]
    if missing:
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY, "Missing required field(s).",
                       {"missing_fields": missing})

    cleaned: Dict[str, Any] = {}
    if "title" in payload:
        cleaned["title"] = _validate_title(payload["title"])
    if "completed" in payload:
        cleaned["completed"] = _validate_completed(payload["completed"])
    return cleaned


# --------------------------------------------------------------------------- #
# Storage
# --------------------------------------------------------------------------- #


class TodoStore:
    """Thread-safe in-memory store with optional atomic JSON-file persistence."""

    def __init__(self, data_file: Optional[str] = None) -> None:
        self._lock = threading.RLock()
        self._todos: Dict[str, Todo] = {}
        self._data_file = os.path.abspath(data_file) if data_file else None
        if self._data_file:
            self._load()

    # -- persistence -------------------------------------------------------- #

    def _load(self) -> None:
        assert self._data_file is not None
        if not os.path.exists(self._data_file):
            LOG.info("Data file %s does not exist yet; starting empty.", self._data_file)
            return
        try:
            with open(self._data_file, "r", encoding="utf-8") as fh:
                raw = json.load(fh)
            items = raw.get("todos", []) if isinstance(raw, dict) else raw
            if not isinstance(items, list):
                raise ValueError("expected a list of todos")
            loaded = {}
            for item in items:
                todo = Todo.from_dict(item)
                loaded[todo.id] = todo
        except (OSError, ValueError, KeyError, TypeError) as exc:
            raise RuntimeError(f"Failed to load data file {self._data_file!r}: {exc}") from exc
        self._todos = loaded
        LOG.info("Loaded %d todo(s) from %s.", len(self._todos), self._data_file)

    def _persist(self) -> None:
        """Write the current state atomically. Caller must hold the lock."""
        if not self._data_file:
            return
        directory = os.path.dirname(self._data_file) or "."
        os.makedirs(directory, exist_ok=True)
        payload = {"version": 1, "todos": [t.to_dict() for t in self._sorted()]}
        fd, tmp_path = tempfile.mkstemp(prefix=".todos-", suffix=".tmp", dir=directory)
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                json.dump(payload, fh, ensure_ascii=False, indent=2)
                fh.flush()
                os.fsync(fh.fileno())
            os.replace(tmp_path, self._data_file)
        except BaseException:
            try:
                os.unlink(tmp_path)
            except OSError:
                pass
            raise

    def _sorted(self) -> List[Todo]:
        return sorted(self._todos.values(), key=lambda t: (t.created_at, t.id))

    def _commit(self, rollback: Callable[[], None]) -> None:
        """Persist; on failure, undo the in-memory change and surface a 500."""
        try:
            self._persist()
        except OSError as exc:
            rollback()
            LOG.exception("Failed to persist data.")
            raise APIError(HTTPStatus.INTERNAL_SERVER_ERROR, "Failed to persist data.") from exc

    # -- operations --------------------------------------------------------- #

    def list(self, completed: Optional[bool] = None) -> List[Todo]:
        with self._lock:
            todos = self._sorted()
        if completed is not None:
            todos = [t for t in todos if t.completed is completed]
        return todos

    def get(self, todo_id: str) -> Todo:
        with self._lock:
            todo = self._todos.get(todo_id)
            if todo is None:
                raise APIError(HTTPStatus.NOT_FOUND, f"Todo {todo_id!r} not found.")
            return Todo(**todo.to_dict())

    def create(self, title: str, completed: bool = False) -> Todo:
        now = utc_now_iso()
        todo = Todo(id=uuid.uuid4().hex, title=title, completed=completed,
                    created_at=now, updated_at=now)
        with self._lock:
            self._todos[todo.id] = todo
            self._commit(lambda: self._todos.pop(todo.id, None))
            return Todo(**todo.to_dict())

    def update(self, todo_id: str, changes: Dict[str, Any]) -> Todo:
        with self._lock:
            existing = self._todos.get(todo_id)
            if existing is None:
                raise APIError(HTTPStatus.NOT_FOUND, f"Todo {todo_id!r} not found.")
            previous = Todo(**existing.to_dict())
            if changes:
                for key, value in changes.items():
                    setattr(existing, key, value)
                existing.updated_at = utc_now_iso()

                def rollback() -> None:
                    self._todos[todo_id] = previous

                self._commit(rollback)
            return Todo(**existing.to_dict())

    def delete(self, todo_id: str) -> None:
        with self._lock:
            removed = self._todos.pop(todo_id, None)
            if removed is None:
                raise APIError(HTTPStatus.NOT_FOUND, f"Todo {todo_id!r} not found.")

            def rollback() -> None:
                self._todos[todo_id] = removed

            self._commit(rollback)


# --------------------------------------------------------------------------- #
# Routing
# --------------------------------------------------------------------------- #

Handler = Callable[["TodoRequestHandler", Dict[str, str]], Tuple[HTTPStatus, Any]]


class Router:
    def __init__(self) -> None:
        self._routes: List[Tuple[re.Pattern, Dict[str, Handler]]] = []

    def add(self, pattern: str, method: str, handler: Handler) -> None:
        regex = re.compile("^" + re.sub(r"\{(\w+)\}", r"(?P<\1>[^/]+)", pattern) + "/?$")
        for existing_regex, methods in self._routes:
            if existing_regex.pattern == regex.pattern:
                methods[method.upper()] = handler
                return
        self._routes.append((regex, {method.upper(): handler}))

    def resolve(self, method: str, path: str) -> Tuple[Handler, Dict[str, str]]:
        for regex, methods in self._routes:
            match = regex.match(path)
            if not match:
                continue
            handler = methods.get(method.upper())
            if handler is None:
                allowed = ", ".join(sorted(methods))
                raise APIError(HTTPStatus.METHOD_NOT_ALLOWED,
                               f"Method {method} not allowed on {path}.",
                               {"allowed_methods": sorted(methods)},
                               headers={"Allow": allowed})
            return handler, match.groupdict()
        raise APIError(HTTPStatus.NOT_FOUND, f"No route for {path}.")

    def allowed_methods(self, path: str) -> Optional[List[str]]:
        for regex, methods in self._routes:
            if regex.match(path):
                return sorted(set(methods) | {"OPTIONS"})
        return None


def parse_bool(value: str, name: str) -> bool:
    lowered = value.strip().lower()
    if lowered in ("true", "1", "yes"):
        return True
    if lowered in ("false", "0", "no"):
        return False
    raise APIError(HTTPStatus.BAD_REQUEST,
                   f"Query parameter {name!r} must be a boolean (true/false).")


# --------------------------------------------------------------------------- #
# Endpoint implementations
# --------------------------------------------------------------------------- #


def handle_health(req: "TodoRequestHandler", _params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    return HTTPStatus.OK, {"status": "ok", "version": API_VERSION, "time": utc_now_iso()}


def handle_list(req: "TodoRequestHandler", _params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    query = req.query
    completed: Optional[bool] = None
    if "completed" in query:
        completed = parse_bool(query["completed"][-1], "completed")
    todos = req.store.list(completed=completed)
    return HTTPStatus.OK, {"todos": [t.to_dict() for t in todos], "count": len(todos)}


def handle_create(req: "TodoRequestHandler", _params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    data = validate_payload(req.read_json(), required=("title",))
    todo = req.store.create(title=data["title"], completed=data.get("completed", False))
    req.extra_headers["Location"] = f"/todos/{todo.id}"
    return HTTPStatus.CREATED, todo.to_dict()


def handle_get(req: "TodoRequestHandler", params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    return HTTPStatus.OK, req.store.get(params["todo_id"]).to_dict()


def handle_replace(req: "TodoRequestHandler", params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    data = validate_payload(req.read_json(), required=("title", "completed"))
    return HTTPStatus.OK, req.store.update(params["todo_id"], data).to_dict()


def handle_patch(req: "TodoRequestHandler", params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    data = validate_payload(req.read_json(), required=())
    if not data:
        raise APIError(HTTPStatus.UNPROCESSABLE_ENTITY,
                       "PATCH body must contain at least one field.",
                       {"allowed_fields": sorted(ALLOWED_FIELDS)})
    return HTTPStatus.OK, req.store.update(params["todo_id"], data).to_dict()


def handle_delete(req: "TodoRequestHandler", params: Dict[str, str]) -> Tuple[HTTPStatus, Any]:
    req.store.delete(params["todo_id"])
    return HTTPStatus.NO_CONTENT, None


def build_router() -> Router:
    router = Router()
    router.add("/health", "GET", handle_health)
    router.add("/todos", "GET", handle_list)
    router.add("/todos", "POST", handle_create)
    router.add("/todos/{todo_id}", "GET", handle_get)
    router.add("/todos/{todo_id}", "PUT", handle_replace)
    router.add("/todos/{todo_id}", "PATCH", handle_patch)
    router.add("/todos/{todo_id}", "DELETE", handle_delete)
    return router


# --------------------------------------------------------------------------- #
# HTTP layer
# --------------------------------------------------------------------------- #


class TodoRequestHandler(BaseHTTPRequestHandler):
    server_version = f"TodoAPI/{API_VERSION}"
    protocol_version = "HTTP/1.1"

    # Injected by TodoHTTPServer.
    store: TodoStore
    router: Router
    cors_origin: Optional[str]

    def setup(self) -> None:
        super().setup()
        self.extra_headers: Dict[str, str] = {}
        self.query: Dict[str, List[str]] = {}

    # -- helpers ------------------------------------------------------------ #

    def read_json(self) -> Any:
        if "chunked" in self.headers.get("Transfer-Encoding", "").lower():
            raise APIError(HTTPStatus.LENGTH_REQUIRED,
                           "Chunked transfer encoding is not supported; send Content-Length.")
        length_header = self.headers.get("Content-Length")
        if length_header is None:
            raise APIError(HTTPStatus.LENGTH_REQUIRED, "Content-Length header is required.")
        try:
            length = int(length_header)
        except ValueError:
            raise APIError(HTTPStatus.BAD_REQUEST, "Invalid Content-Length header.") from None
        if length < 0:
            raise APIError(HTTPStatus.BAD_REQUEST, "Invalid Content-Length header.")
        if length > MAX_BODY_BYTES:
            self.close_connection = True
            raise APIError(HTTPStatus.REQUEST_ENTITY_TOO_LARGE,
                           f"Request body exceeds {MAX_BODY_BYTES} bytes.",
                           {"max_bytes": MAX_BODY_BYTES})

        raw = self.rfile.read(length) if length else b""

        content_type = self.headers.get("Content-Type", "")
        media_type = content_type.split(";", 1)[0].strip().lower()
        if media_type != "application/json":
            raise APIError(HTTPStatus.UNSUPPORTED_MEDIA_TYPE,
                           "Content-Type must be application/json.")
        if not raw:
            raise APIError(HTTPStatus.BAD_REQUEST, "Request body must not be empty.")
        try:
            return json.loads(raw.decode("utf-8"))
        except UnicodeDecodeError:
            raise APIError(HTTPStatus.BAD_REQUEST, "Request body must be UTF-8 encoded.") from None
        except json.JSONDecodeError as exc:
            raise APIError(HTTPStatus.BAD_REQUEST, "Malformed JSON in request body.",
                           {"line": exc.lineno, "column": exc.colno, "reason": exc.msg}) from None

    def _drain_body(self) -> None:
        """Discard any unread request body so keep-alive connections stay in sync."""
        try:
            length = int(self.headers.get("Content-Length", "0") or 0)
        except ValueError:
            self.close_connection = True
            return
        if 0 < length <= MAX_BODY_BYTES:
            self.rfile.read(length)
        elif length > MAX_BODY_BYTES:
            self.close_connection = True

    def _send_common_headers(self) -> None:
        self.send_header("X-Content-Type-Options", "nosniff")
        self.send_header("Cache-Control", "no-store")
        if self.cors_origin:
            self.send_header("Access-Control-Allow-Origin", self.cors_origin)
            self.send_header("Vary", "Origin")
        for key, value in self.extra_headers.items():
            self.send_header(key, value)

    def _send_json(self, status: HTTPStatus, body: Any) -> None:
        if status == HTTPStatus.NO_CONTENT or body is None:
            self.send_response(status)
            self._send_common_headers()
            self.send_header("Content-Length", "0")
            self.end_headers()
            return
        encoded = json.dumps(body, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
        self.send_response(status)
        self._send_common_headers()
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(encoded)))
        self.end_headers()
        if self.command != "HEAD":
            self.wfile.write(encoded)

    # -- dispatch ----------------------------------------------------------- #

    def _dispatch(self) -> None:
        body_consumed_by_handler = self.command in ("POST", "PUT", "PATCH")
        try:
            parts = urlsplit(self.path)
            self.query = parse_qs(parts.query, keep_blank_values=True)
            method = "GET" if self.command == "HEAD" else self.command
            handler, params = self.router.resolve(method, parts.path)
            status, body = handler(self, params)
            if not body_consumed_by_handler:
                self._drain_body()
            self._send_json(status, body)
        except APIError as err:
            if not body_consumed_by_handler:
                self._drain_body()
            self.extra_headers.update(err.headers)
            self._send_json(err.status, err.to_dict())
        except (BrokenPipeError, ConnectionResetError):
            LOG.debug("Client disconnected during request.")
            self.close_connection = True
        except Exception:  # noqa: BLE001 - last-resort safety net
            LOG.exception("Unhandled error processing %s %s", self.command, self.path)
            self.close_connection = True
            err = APIError(HTTPStatus.INTERNAL_SERVER_ERROR, "Internal server error.")
            try:
                self._send_json(err.status, err.to_dict())
            except OSError:
                pass

    def do_GET(self) -> None:  # noqa: N802
        self._dispatch()

    def do_HEAD(self) -> None:  # noqa: N802
        self._dispatch()

    def do_POST(self) -> None:  # noqa: N802
        self._dispatch()

    def do_PUT(self) -> None:  # noqa: N802
        self._dispatch()

    def do_PATCH(self) -> None:  # noqa: N802
        self._dispatch()

    def do_DELETE(self) -> None:  # noqa: N802
        self._dispatch()

    def do_OPTIONS(self) -> None:  # noqa: N802
        self._drain_body()
        path = urlsplit(self.path).path
        allowed = self.router.allowed_methods(path)
        if allowed is None:
            err = APIError(HTTPStatus.NOT_FOUND, f"No route for {path}.")
            self._send_json(err.status, err.to_dict())
            return
        if "GET" in allowed:
            allowed = sorted(set(allowed) | {"HEAD"})
        allow = ", ".join(allowed)
        self.extra_headers["Allow"] = allow
        if self.cors_origin:
            self.extra_headers["Access-Control-Allow-Methods"] = allow
            self.extra_headers["Access-Control-Allow-Headers"] = "Content-Type"
            self.extra_headers["Access-Control-Max-Age"] = "600"
        self._send_json(HTTPStatus.NO_CONTENT, None)

    # -- logging ------------------------------------------------------------ #

    def log_message(self, format: str, *args: Any) -> None:  # noqa: A002
        LOG.info("%s - %s", self.address_string(), format % args)

    def log_error(self, format: str, *args: Any) -> None:  # noqa: A002
        LOG.warning("%s - %s", self.address_string(), format % args)


class TodoHTTPServer(ThreadingHTTPServer):
    daemon_threads = True
    allow_reuse_address = True

    def __init__(self, address: Tuple[str, int], store: TodoStore,
                 cors_origin: Optional[str] = None) -> None:
        handler_cls = type(
            "BoundTodoRequestHandler",
            (TodoRequestHandler,),
            {"store": store, "router": build_router(), "cors_origin": cors_origin},
        )
        super().__init__(address, handler_cls)


# --------------------------------------------------------------------------- #
# Entrypoint
# --------------------------------------------------------------------------- #


def parse_args(argv: Optional[List[str]] = None) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Serve a small JSON to-do list API.")
    parser.add_argument("--host", default=os.environ.get("TODO_HOST", "127.0.0.1"),
                        help="Interface to bind (default: 127.0.0.1, env TODO_HOST).")
    parser.add_argument("--port", type=int, default=int(os.environ.get("TODO_PORT", "8000")),
                        help="Port to listen on (default: 8000, env TODO_PORT).")
    parser.add_argument("--data-file", default=os.environ.get("TODO_DATA_FILE"),
                        help="Optional JSON file for persistence (env TODO_DATA_FILE).")
    parser.add_argument("--cors-origin", default=os.environ.get("TODO_CORS_ORIGIN"),
                        help="Value for Access-Control-Allow-Origin; disabled if unset.")
    parser.add_argument("--log-level", default=os.environ.get("TODO_LOG_LEVEL", "INFO"),
                        choices=["DEBUG", "INFO", "WARNING", "ERROR"],
                        help="Logging verbosity (default: INFO).")
    args = parser.parse_args(argv)
    if not (0 <= args.port <= 65535):
        parser.error("--port must be between 0 and 65535")
    return args


def main(argv: Optional[List[str]] = None) -> int:
    args = parse_args(argv)
    logging.basicConfig(
        level=getattr(logging, args.log_level),
        format="%(asctime)s %(levelname)s [%(threadName)s] %(name)s: %(message)s",
    )

    try:
        store = TodoStore(args.data_file)
    except RuntimeError as exc:
        LOG.error("%s", exc)
        return 1

    try:
        server = TodoHTTPServer((args.host, args.port), store, cors_origin=args.cors_origin)
    except OSError as exc:
        LOG.error("Could not bind to %s:%d: %s", args.host, args.port, exc)
        return 1

    def shutdown_handler(signum: int, _frame: Any) -> None:
        LOG.info("Received signal %s; shutting down.", signal.Signals(signum).name)
        # shutdown() blocks until serve_forever exits, so call it from another thread.
        threading.Thread(target=server.shutdown, daemon=True).start()

    signal.signal(signal.SIGINT, shutdown_handler)
    signal.signal(signal.SIGTERM, shutdown_handler)

    host, port = server.server_address[:2]
    LOG.info("Todo API listening on http://%s:%d (persistence: %s)",
             host, port, args.data_file or "in-memory")
    try:
        server.serve_forever(poll_interval=0.5)
    finally:
        server.server_close()
        LOG.info("Server stopped.")
    return 0


if __name__ == "__main__":
    sys.exit(main())

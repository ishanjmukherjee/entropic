-- =============================================================================
-- Library Lending System (PostgreSQL)
-- Schema, checkout/return procedures, and overdue reporting.
-- =============================================================================

BEGIN;

CREATE SCHEMA IF NOT EXISTS library;
SET search_path TO library, public;

-- -----------------------------------------------------------------------------
-- Reference data: membership types govern loan limits, periods and fines.
-- -----------------------------------------------------------------------------
CREATE TABLE membership_types (
    code              text          PRIMARY KEY,
    max_loans         integer       NOT NULL CHECK (max_loans > 0),
    loan_period_days  integer       NOT NULL CHECK (loan_period_days > 0),
    fine_per_day      numeric(8,2)  NOT NULL DEFAULT 0 CHECK (fine_per_day >= 0)
);

-- -----------------------------------------------------------------------------
-- Members
-- -----------------------------------------------------------------------------
CREATE TABLE members (
    member_id        bigint       GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    full_name        text         NOT NULL CHECK (length(btrim(full_name)) > 0),
    email            text         NOT NULL CHECK (email LIKE '%_@_%'),
    membership_type  text         NOT NULL REFERENCES membership_types (code),
    status           text         NOT NULL DEFAULT 'active'
                                  CHECK (status IN ('active', 'suspended', 'expired')),
    joined_on        date         NOT NULL DEFAULT current_date,
    created_at       timestamptz  NOT NULL DEFAULT now()
);

CREATE UNIQUE INDEX members_email_key ON members (lower(email));

-- -----------------------------------------------------------------------------
-- Catalogue: titles and physical copies
-- -----------------------------------------------------------------------------
CREATE TABLE books (
    book_id          bigint       GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    isbn             text         UNIQUE
                                  CHECK (isbn IS NULL OR isbn ~ '^[0-9]{10}([0-9]{3})?$'),
    title            text         NOT NULL CHECK (length(btrim(title)) > 0),
    author           text         NOT NULL,
    published_year   smallint     CHECK (published_year BETWEEN 1400 AND 2100),
    created_at       timestamptz  NOT NULL DEFAULT now()
);

CREATE TABLE book_copies (
    copy_id      bigint       GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    book_id      bigint       NOT NULL REFERENCES books (book_id),
    barcode      text         NOT NULL UNIQUE CHECK (length(btrim(barcode)) > 0),
    status       text         NOT NULL DEFAULT 'available'
                              CHECK (status IN ('available', 'on_loan', 'maintenance', 'lost', 'withdrawn')),
    acquired_on  date         NOT NULL DEFAULT current_date
);

CREATE INDEX book_copies_book_id_idx ON book_copies (book_id);

-- -----------------------------------------------------------------------------
-- Loans. Each row is one checkout; returned_at IS NULL means still out.
-- -----------------------------------------------------------------------------
CREATE TABLE loans (
    loan_id     bigint       GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    copy_id     bigint       NOT NULL REFERENCES book_copies (copy_id),
    member_id   bigint       NOT NULL REFERENCES members (member_id),
    loaned_at   timestamptz  NOT NULL DEFAULT now(),
    due_date    date         NOT NULL,
    returned_at timestamptz,
    CONSTRAINT loans_return_after_loan CHECK (returned_at IS NULL OR returned_at >= loaned_at)
);

-- A copy can be out on at most one loan at a time.
CREATE UNIQUE INDEX loans_one_open_per_copy ON loans (copy_id) WHERE returned_at IS NULL;
-- Fast lookup of a member's open loans and of overdue loans.
CREATE INDEX loans_open_by_member ON loans (member_id) WHERE returned_at IS NULL;
CREATE INDEX loans_open_due_date  ON loans (due_date) WHERE returned_at IS NULL;
-- Loan history per member.
CREATE INDEX loans_member_history ON loans (member_id, loaned_at);

-- -----------------------------------------------------------------------------
-- Fines assessed on late returns.
-- -----------------------------------------------------------------------------
CREATE TABLE fines (
    fine_id      bigint       GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    loan_id      bigint       NOT NULL UNIQUE REFERENCES loans (loan_id),
    amount       numeric(8,2) NOT NULL CHECK (amount > 0),
    assessed_at  timestamptz  NOT NULL DEFAULT now(),
    paid_at      timestamptz,
    CONSTRAINT fines_paid_after_assessed CHECK (paid_at IS NULL OR paid_at >= assessed_at)
);

CREATE INDEX fines_unpaid ON fines (loan_id) WHERE paid_at IS NULL;

-- -----------------------------------------------------------------------------
-- Seed data
-- -----------------------------------------------------------------------------
INSERT INTO membership_types (code, max_loans, loan_period_days, fine_per_day)
VALUES
    ('standard', 5,  21, 0.25),
    ('student',  10, 14, 0.10),
    ('senior',   8,  28, 0.00)
ON CONFLICT (code) DO NOTHING;

-- =============================================================================
-- Checkout
-- Returns the new loan_id. Raises an exception if the checkout is not allowed.
-- =============================================================================
CREATE OR REPLACE FUNCTION checkout_copy(p_member_id bigint, p_barcode text)
RETURNS bigint
LANGUAGE plpgsql
AS $$
#variable_conflict use_column
DECLARE
    v_member_status  text;
    v_membership     text;
    v_max_loans      integer;
    v_period_days    integer;
    v_open_loans     integer;
    v_copy_id        bigint;
    v_copy_status    text;
    v_loan_id        bigint;
    v_today          date := current_date;
BEGIN
    -- Lock the member row to serialise concurrent checkouts for the same member.
    SELECT m.status, m.membership_type
      INTO v_member_status, v_membership
      FROM members m
     WHERE m.member_id = p_member_id
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'member % does not exist', p_member_id
            USING ERRCODE = 'no_data_found';
    END IF;

    IF v_member_status <> 'active' THEN
        RAISE EXCEPTION 'member % is not active (status: %)', p_member_id, v_member_status
            USING ERRCODE = 'check_violation';
    END IF;

    SELECT mt.max_loans, mt.loan_period_days
      INTO v_max_loans, v_period_days
      FROM membership_types mt
     WHERE mt.code = v_membership;

    IF EXISTS (
        SELECT 1
          FROM loans l
         WHERE l.member_id = p_member_id
           AND l.returned_at IS NULL
           AND l.due_date < v_today
    ) THEN
        RAISE EXCEPTION 'member % has overdue items', p_member_id
            USING ERRCODE = 'check_violation';
    END IF;

    IF EXISTS (
        SELECT 1
          FROM fines f
          JOIN loans l ON l.loan_id = f.loan_id
         WHERE l.member_id = p_member_id
           AND f.paid_at IS NULL
    ) THEN
        RAISE EXCEPTION 'member % has unpaid fines', p_member_id
            USING ERRCODE = 'check_violation';
    END IF;

    SELECT count(*)
      INTO v_open_loans
      FROM loans l
     WHERE l.member_id = p_member_id
       AND l.returned_at IS NULL;

    IF v_open_loans >= v_max_loans THEN
        RAISE EXCEPTION 'member % has reached the loan limit of %', p_member_id, v_max_loans
            USING ERRCODE = 'check_violation';
    END IF;

    -- Lock the copy so two desks cannot check out the same item.
    SELECT c.copy_id, c.status
      INTO v_copy_id, v_copy_status
      FROM book_copies c
     WHERE c.barcode = p_barcode
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'no copy with barcode %', p_barcode
            USING ERRCODE = 'no_data_found';
    END IF;

    IF v_copy_status <> 'available' THEN
        RAISE EXCEPTION 'copy % is not available (status: %)', p_barcode, v_copy_status
            USING ERRCODE = 'check_violation';
    END IF;

    INSERT INTO loans (copy_id, member_id, loaned_at, due_date)
    VALUES (v_copy_id, p_member_id, now(), v_today + v_period_days)
    RETURNING loans.loan_id INTO v_loan_id;

    UPDATE book_copies
       SET status = 'on_loan'
     WHERE book_copies.copy_id = v_copy_id;

    RETURN v_loan_id;
END;
$$;

-- =============================================================================
-- Return
-- Closes the open loan for a barcode, assesses any late fine, and makes the
-- copy available again. Returns the loan, days overdue, and fine assessed.
-- =============================================================================
CREATE OR REPLACE FUNCTION return_copy(p_barcode text)
RETURNS TABLE (
    loan_id      bigint,
    days_overdue integer,
    fine_amount  numeric(8,2)
)
LANGUAGE plpgsql
AS $$
#variable_conflict use_column
DECLARE
    v_copy_id  bigint;
    v_loan     loans%ROWTYPE;
    v_rate     numeric(8,2);
    v_today    date := current_date;
    v_days     integer;
    v_fine     numeric(8,2) := 0;
BEGIN
    SELECT c.copy_id
      INTO v_copy_id
      FROM book_copies c
     WHERE c.barcode = p_barcode
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'no copy with barcode %', p_barcode
            USING ERRCODE = 'no_data_found';
    END IF;

    SELECT l.*
      INTO v_loan
      FROM loans l
     WHERE l.copy_id = v_copy_id
       AND l.returned_at IS NULL
       FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'copy % is not currently on loan', p_barcode
            USING ERRCODE = 'check_violation';
    END IF;

    UPDATE loans
       SET returned_at = now()
     WHERE loans.loan_id = v_loan.loan_id;

    v_days := GREATEST(0, v_today - v_loan.due_date);

    SELECT mt.fine_per_day
      INTO v_rate
      FROM members m
      JOIN membership_types mt ON mt.code = m.membership_type
     WHERE m.member_id = v_loan.member_id;

    IF v_days > 0 AND v_rate > 0 THEN
        v_fine := round(v_days * v_rate, 2);

        INSERT INTO fines (loan_id, amount)
        VALUES (v_loan.loan_id, v_fine);
    END IF;

    UPDATE book_copies
       SET status = 'available'
     WHERE book_copies.copy_id = v_copy_id
       AND book_copies.status = 'on_loan';

    RETURN QUERY SELECT v_loan.loan_id, v_days, v_fine;
END;
$$;

-- =============================================================================
-- Overdue report
-- Lists every open loan past its due date as of p_as_of, with days overdue
-- and the fine that would accrue at that date.
-- =============================================================================
CREATE OR REPLACE FUNCTION overdue_loans(p_as_of date DEFAULT current_date)
RETURNS TABLE (
    loan_id          bigint,
    member_id        bigint,
    member_name      text,
    member_email     text,
    barcode          text,
    title            text,
    loaned_at        timestamptz,
    due_date         date,
    days_overdue     integer,
    estimated_fine   numeric(8,2)
)
LANGUAGE sql
STABLE
AS $$
    SELECT l.loan_id,
           m.member_id,
           m.full_name,
           m.email,
           c.barcode,
           b.title,
           l.loaned_at,
           l.due_date,
           (p_as_of - l.due_date)::integer                    AS days_overdue,
           round((p_as_of - l.due_date) * mt.fine_per_day, 2) AS estimated_fine
      FROM loans l
      JOIN members          m  ON m.member_id = l.member_id
      JOIN membership_types mt ON mt.code     = m.membership_type
      JOIN book_copies      c  ON c.copy_id   = l.copy_id
      JOIN books            b  ON b.book_id   = c.book_id
     WHERE l.returned_at IS NULL
       AND l.due_date < p_as_of
     ORDER BY (p_as_of - l.due_date) DESC, l.due_date, l.loan_id;
$$;

-- Summary of overdue activity per member as of a given date.
CREATE OR REPLACE FUNCTION overdue_summary(p_as_of date DEFAULT current_date)
RETURNS TABLE (
    member_id       bigint,
    member_name     text,
    overdue_items  integer,
    max_days_overdue integer,
    total_estimated_fine numeric(10,2)
)
LANGUAGE sql
STABLE
AS $$
    SELECT o.member_id,
           o.member_name,
           count(*)::integer,
           max(o.days_overdue),
           sum(o.estimated_fine)
      FROM overdue_loans(p_as_of) o
     GROUP BY o.member_id, o.member_name
     ORDER BY max(o.days_overdue) DESC, o.member_name;
$$;

COMMIT;

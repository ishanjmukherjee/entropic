use std::fmt;

/// Represents the different kinds of tokens in our arithmetic expressions.
#[derive(Debug, Clone, PartialEq)]
pub enum TokenKind {
    /// A floating-point number literal
    Number(f64),
    /// A variable identifier (starts with letter or underscore,
    /// followed by letters, digits, underscores)
    Ident(String),
    /// '+'
    Plus,
    /// '-'
    Minus,
    /// '*'
    Star,
    /// '/'
    Slash,
    /// '^'
    Caret,
    /// '('
    LParen,
    /// ')'
    RParen,
    /// End of input
    Eof,
}

/// A token produced by the lexer, carrying its kind and source location.
#[derive(Debug, Clone)]
pub struct Token {
    pub kind: TokenKind,
    pub offset: usize,
}

/// The lexer (tokenizer). It processes the input string and yields tokens on demand.
pub struct Tokenizer {
    input: Vec<char>,
    pos: usize,
}

impl Tokenizer {
    pub fn new(source: &str) -> Self {
        Self {
            input: source.chars().collect(),
            pos: 0,
        }
    }

    /// Returns the current character, if any.
    fn current_char(&self) -> Option<char> {
        self.input.get(self.pos).copied()
    }

    /// Advances the position by one and returns the previous character.
    fn advance(&mut self) -> Option<char> {
        let ch = self.current_char();
        if ch.is_some() {
            self.pos += 1;
        }
        ch
    }

    /// Skips whitespace characters.
    fn skip_whitespace(&mut self) {
        while let Some(ch) = self.current_char() {
            if ch.is_ascii_whitespace() {
                self.pos += 1;
            } else {
                break;
            }
        }
    }

    /// Reads a numeric literal. Assumes the current character is a digit or a dot
    /// followed by a digit.
    fn read_number(&mut self) -> TokenKind {
        let start = self.pos;
        let mut has_dot = false;

        while let Some(ch) = self.current_char() {
            if ch.is_ascii_digit() {
                self.pos += 1;
            } else if ch == '.' && !has_dot {
                // Peek ahead to ensure it's not a lone dot (e.g., "x." is invalid)
                let next = self.input.get(self.pos + 1).copied();
                if let Some(nch) = next {
                    if nch.is_ascii_digit() {
                        has_dot = true;
                        self.pos += 1; // consume the dot
                    } else {
                        break; // treat dot as separate token
                    }
                } else {
                    break;
                }
            } else {
                break;
            }
        }

        let lexeme: String = self.input[start..self.pos].iter().collect();
        let value: f64 = lexeme.parse().expect("invalid number format");
        TokenKind::Number(value)
    }

    /// Reads an identifier. Assumes the current character is a letter or underscore.
    fn read_ident(&mut self) -> TokenKind {
        let start = self.pos;
        while let Some(ch) = self.current_char() {
            if ch.is_alphanumeric() || ch == '_' {
                self.pos += 1;
            } else {
                break;
            }
        }
        let name: String = self.input[start..self.pos].iter().collect();
        TokenKind::Ident(name)
    }

    /// Returns the next token without consuming it (lookahead).
    pub fn peek_token(&mut self) -> Token {
        let saved_pos = self.pos;
        let token = self.next_token();
        self.pos = saved_pos;
        token
    }

    /// Consumes and returns the next token from the input.
    pub fn next_token(&mut self) -> Token {
        self.skip_whitespace();

        let offset = self.pos;

        let kind = match self.current_char() {
            None => TokenKind::Eof,
            Some(ch) if ch.is_ascii_digit() => self.read_number(),
            Some(ch) if ch.is_alphabetic() || ch == '_' => self.read_ident(),
            Some('+') => {
                self.pos += 1;
                TokenKind::Plus
            }
            Some('-') => {
                self.pos += 1;
                TokenKind::Minus
            }
            Some('*') => {
                self.pos += 1;
                TokenKind::Star
            }
            Some('/') => {
                self.pos += 1;
                TokenKind::Slash
            }
            Some('^') => {
                self.pos += 1;
                TokenKind::Caret
            }
            Some('(') => {
                self.pos += 1;
                TokenKind::LParen
            }
            Some(')') => {
                self.pos += 1;
                TokenKind::RParen
            }
            Some(ch) => {
                // Unknown character – we simply skip it and return Eof to stop parsing.
                // In production code you might want to emit an error token.
                self.pos += 1;
                eprintln!("Warning: ignoring unexpected character '{}' at position {}", ch, offset);
                TokenKind::Eof
            }
        };

        Token { kind, offset }
    }
}

/// AST node types for arithmetic expressions.
#[derive(Debug, Clone, PartialEq)]
pub enum Expr {
    /// A numeric literal
    Number(f64),
    /// A variable reference
    Variable(String),
    /// Binary operation: left operand, operator, right operand
    Binary {
        left: Box<Expr>,
        op: BinOp,
        right: Box<Expr>,
    },
    /// Unary minus (e.g., -x, -(a + b))
    UnaryMinus(Box<Expr>),
}

/// Binary operators supported in expressions.
#[derive(Debug, Clone, Copy, PartialEq)]
pub enum BinOp {
    Add,
    Sub,
    Mul,
    Div,
    Pow,
}

impl fmt::Display for BinOp {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            BinOp::Add => write!(f, "+"),
            BinOp::Sub => write!(f, "-"),
            BinOp::Mul => write!(f, "*"),
            BinOp::Div => write!(f, "/"),
            BinOp::Pow => write!(f, "^"),
        }
    }
}

/// Custom error type for parsing failures.
#[derive(Debug, Clone)]
pub struct ParseError {
    pub message: String,
    pub offset: usize,
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "Parse error at position {}: {}", self.offset, self.message)
    }
}

/// The recursive-descent parser.
pub struct Parser {
    tokenizer: Tokenizer,
    /// Current lookahead token
    current: Token,
}

impl Parser {
    pub fn new(source: &str) -> Self {
        let mut tokenizer = Tokenizer::new(source);
        let current = tokenizer.next_token();
        Parser { tokenizer, current }
    }

    /// Consumes the current token if its kind matches `expected`, otherwise returns an error.
    fn expect(&mut self, expected: TokenKind) -> Result<(), ParseError> {
        if self.current.kind == expected {
            self.current = self.tokenizer.next_token();
            Ok(())
        } else {
            Err(ParseError {
                message: format!(
                    "expected {:?}, found {:?}",
                    expected, self.current.kind
                ),
                offset: self.current.offset,
            })
        }
    }

    /// Entry point: parse a complete expression. Ensures we consume all input.
    pub fn parse_expression(&mut self) -> Result<Expr, ParseError> {
        let expr = self.parse_add_sub()?;
        if self.current.kind != TokenKind::Eof {
            return Err(ParseError {
                message: format!("unexpected token {:?} after expression", self.current.kind),
                offset: self.current.offset,
            });
        }
        Ok(expr)
    }

    /// Parses addition and subtraction (lowest precedence).
    fn parse_add_sub(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_mul_div()?;

        loop {
            match self.current.kind {
                TokenKind::Plus => {
                    self.current = self.tokenizer.next_token();
                    let right = self.parse_mul_div()?;
                    left = Expr::Binary {
                        left: Box::new(left),
                        op: BinOp::Add,
                        right: Box::new(right),
                    };
                }
                TokenKind::Minus => {
                    self.current = self.tokenizer.next_token();
                    let right = self.parse_mul_div()?;
                    left = Expr::Binary {
                        left: Box::new(left),
                        op: BinOp::Sub,
                        right: Box::new(right),
                    };
                }
                _ => break,
            }
        }
        Ok(left)
    }

    /// Parses multiplication and division (medium precedence).
    fn parse_mul_div(&mut self) -> Result<Expr, ParseError> {
        let mut left = self.parse_power()?;

        loop {
            match self.current.kind {
                TokenKind::Star => {
                    self.current = self.tokenizer.next_token();
                    let right = self.parse_power()?;
                    left = Expr::Binary {
                        left: Box::new(left),
                        op: BinOp::Mul,
                        right: Box::new(right),
                    };
                }
                TokenKind::Slash => {
                    self.current = self.tokenizer.next_token();
                    let right = self.parse_power()?;
                    left = Expr::Binary {
                        left: Box::new(left),
                        op: BinOp::Div,
                        right: Box::new(right),
                    };
                }
                _ => break,
            }
        }
        Ok(left)
    }

    /// Parses exponentiation (high precedence, right‑associative).
    fn parse_power(&mut self) -> Result<Expr, ParseError> {
        let base = self.parse_unary()?;
        if self.current.kind == TokenKind::Caret {
            self.current = self.tokenizer.next_token();
            let exponent = self.parse_power()?; // right‑associative
            Ok(Expr::Binary {
                left: Box::new(base),
                op: BinOp::Pow,
                right: Box::new(exponent),
            })
        } else {
            Ok(base)
        }
    }

    /// Parses unary minus and primary expressions.
    fn parse_unary(&mut self) -> Result<Expr, ParseError> {
        if self.current.kind == TokenKind::Minus {
            let offset = self.current.offset;
            self.current = self.tokenizer.next_token();
            let operand = self.parse_unary()?;
            return Ok(Expr::UnaryMinus(Box::new(operand)));
        }
        self.parse_primary()
    }

    /// Parses primary expressions: numbers, variables, parenthesized expressions.
    fn parse_primary(&mut self) -> Result<Expr, ParseError> {
        match &self.current.kind {
            TokenKind::Number(val) => {
                let expr = Expr::Number(*val);
                self.current = self.tokenizer.next_token();
                Ok(expr)
            }
            TokenKind::Ident(name) => {
                let expr = Expr::Variable(name.clone());
                self.current = self.tokenizer.next_token();
                Ok(expr)
            }
            TokenKind::LParen => {
                self.current = self.tokenizer.next_token(); // consume '('
                let expr = self.parse_add_sub()?;
                self.expect(TokenKind::RParen)?;
                Ok(expr)
            }
            _ => Err(ParseError {
                message: format!("unexpected token {:?}", self.current.kind),
                offset: self.current.offset,
            }),
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn parse(input: &str) -> Result<Expr, ParseError> {
        Parser::new(input).parse_expression()
    }

    #[test]
    fn test_simple_number() {
        assert_eq!(parse("42").unwrap(), Expr::Number(42.0));
    }

    #[test]
    fn test_variable() {
        assert_eq!(parse("x").unwrap(), Expr::Variable("x".into()));
    }

    #[test]
    fn test_addition() {
        assert_eq!(
            parse("a + b").unwrap(),
            Expr::Binary {
                left: Box::new(Expr::Variable("a".into())),
                op: BinOp::Add,
                right: Box::new(Expr::Variable("b".into())),
            }
        );
    }

    #[test]
    fn test_precedence() {
        // 1 + 2 * 3  => 1 + (2 * 3)
        let ast = parse("1 + 2 * 3").unwrap();
        assert_eq!(
            ast,
            Expr::Binary {
                left: Box::new(Expr::Number(1.0)),
                op: BinOp::Add,
                right: Box::new(Expr::Binary {
                    left: Box::new(Expr::Number(2.0)),
                    op: BinOp::Mul,
                    right: Box::new(Expr::Number(3.0)),
                }),
            }
        );
    }

//! Tokenizer and recursive-descent parser for arithmetic expressions with variables.
//!
//! Grammar (precedence low to high):
//!   expr   := term (('+' | '-') term)*
//!   term   := unary (('*' | '/') unary)*
//!   unary  := '-' unary | primary
//!   primary := NUMBER | IDENT | '(' expr ')'
//!
//! Identifiers: [A-Za-z_][A-Za-z0-9_]*
//! Numbers: decimal integers or floats (optional fraction, no exponent).

use std::fmt;

/// Source span in byte offsets (half-open).
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

impl Span {
    pub fn new(start: usize, end: usize) -> Self {
        Self { start, end }
    }
}

#[derive(Debug, Clone, PartialEq)]
pub enum TokenKind {
    Number(f64),
    Ident(String),
    Plus,
    Minus,
    Star,
    Slash,
    LParen,
    RParen,
    Eof,
}

#[derive(Debug, Clone, PartialEq)]
pub struct Token {
    pub kind: TokenKind,
    pub span: Span,
}

#[derive(Debug, Clone, PartialEq)]
pub enum Expr {
    Number(f64, Span),
    Var(String, Span),
    UnaryNeg(Box<Expr>, Span),
    Binary {
        op: BinOp,
        lhs: Box<Expr>,
        rhs: Box<Expr>,
        span: Span,
    },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum BinOp {
    Add,
    Sub,
    Mul,
    Div,
}

impl fmt::Display for BinOp {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.write_str(match self {
            BinOp::Add => "+",
            BinOp::Sub => "-",
            BinOp::Mul => "*",
            BinOp::Div => "/",
        })
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ParseError {
    pub message: String,
    pub span: Span,
}

impl fmt::Display for ParseError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(
            f,
            "{} at bytes {}..{}",
            self.message, self.span.start, self.span.end
        )
    }
}

impl std::error::Error for ParseError {}

pub type ParseResult<T> = Result<T, ParseError>;

/// Lexical scanner. Skips ASCII whitespace. Does not allocate except for identifiers.
pub struct Lexer<'a> {
    src: &'a str,
    bytes: &'a [u8],
    pos: usize,
}

impl<'a> Lexer<'a> {
    pub fn new(src: &'a str) -> Self {
        Self {
            src,
            bytes: src.as_bytes(),
            pos: 0,
        }
    }

    pub fn tokenize(mut self) -> ParseResult<Vec<Token>> {
        let mut out = Vec::new();
        loop {
            let tok = self.next_token()?;
            let is_eof = matches!(tok.kind, TokenKind::Eof);
            out.push(tok);
            if is_eof {
                break;
            }
        }
        Ok(out)
    }

    fn peek(&self) -> Option<u8> {
        self.bytes.get(self.pos).copied()
    }

    fn bump(&mut self) -> Option<u8> {
        let b = self.peek()?;
        self.pos += 1;
        Some(b)
    }

    fn skip_ws(&mut self) {
        while matches!(self.peek(), Some(b) if b.is_ascii_whitespace()) {
            self.pos += 1;
        }
    }

    fn next_token(&mut self) -> ParseResult<Token> {
        self.skip_ws();
        let start = self.pos;
        let Some(b) = self.bump() else {
            return Ok(Token {
                kind: TokenKind::Eof,
                span: Span::new(start, start),
            });
        };

        let kind = match b {
            b'+' => TokenKind::Plus,
            b'-' => TokenKind::Minus,
            b'*' => TokenKind::Star,
            b'/' => TokenKind::Slash,
            b'(' => TokenKind::LParen,
            b')' => TokenKind::RParen,
            b'0'..=b'9' => self.lex_number(start, b)?,
            b'A'..=b'Z' | b'a'..=b'z' | b'_' => self.lex_ident(start),
            _ => {
                return Err(ParseError {
                    message: format!("unexpected character {:?}", b as char),
                    span: Span::new(start, self.pos),
                });
            }
        };

        Ok(Token {
            kind,
            span: Span::new(start, self.pos),
        })
    }

    fn lex_ident(&mut self, start: usize) -> TokenKind {
        while matches!(
            self.peek(),
            Some(b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'_')
        ) {
            self.pos += 1;
        }
        TokenKind::Ident(self.src[start..self.pos].to_string())
    }

    fn lex_number(&mut self, start: usize, first: u8) -> ParseResult<TokenKind> {
        let mut saw_dot = false;
        // Leading zeros are allowed (e.g. 007, 0.5).
        let _ = first;
        while let Some(b) = self.peek() {
            match b {
                b'0'..=b'9' => {
                    self.pos += 1;
                }
                b'.' if !saw_dot => {
                    // Require a digit after the dot so "1." is rejected rather than silently truncated.
                    let next = self.bytes.get(self.pos + 1).copied();
                    if !matches!(next, Some(b'0'..=b'9')) {
                        return Err(ParseError {
                            message: "expected digit after decimal point".into(),
                            span: Span::new(self.pos, self.pos + 1),
                        });
                    }
                    saw_dot = true;
                    self.pos += 1;
                }
                _ => break,
            }
        }
        let text = &self.src[start..self.pos];
        let value: f64 = text.parse().map_err(|_| ParseError {
            message: format!("invalid number `{text}`"),
            span: Span::new(start, self.pos),
        })?;
        if !value.is_finite() {
            return Err(ParseError {
                message: format!("non-finite number `{text}`"),
                span: Span::new(start, self.pos),
            });
        }
        Ok(TokenKind::Number(value))
    }
}

/// Recursive-descent parser over a token slice. Consumes one expression and requires EOF.
pub struct Parser<'a> {
    tokens: &'a [Token],
    idx: usize,
}

impl<'a> Parser<'a> {
    pub fn new(tokens: &'a [Token]) -> Self {
        Self { tokens, idx: 0 }
    }

    pub fn parse_expression(src: &str) -> ParseResult<Expr> {
        let tokens = Lexer::new(src).tokenize()?;
        let mut parser = Parser::new(&tokens);
        let expr = parser.parse_expr()?;
        parser.expect_eof()?;
        Ok(expr)
    }

    fn current(&self) -> &Token {
        self.tokens
            .get(self.idx)
            .expect("token stream always ends with Eof")
    }

    fn advance(&mut self) -> &Token {
        let tok = &self.tokens[self.idx];
        if !matches!(tok.kind, TokenKind::Eof) {
            self.idx += 1;
        }
        tok
    }

    fn expect_eof(&self) -> ParseResult<()> {
        match &self.current().kind {
            TokenKind::Eof => Ok(()),
            other => Err(ParseError {
                message: format!("unexpected token after expression: {other:?}"),
                span: self.current().span,
            }),
        }
    }

    fn parse_expr(&mut self) -> ParseResult<Expr> {
        let mut lhs = self.parse_term()?;
        loop {
            let op = match self.current().kind {
                TokenKind::Plus => BinOp::Add,
                TokenKind::Minus => BinOp::Sub,
                _ => break,
            };
            let op_span = self.current().span;
            self.advance();
            let rhs = self.parse_term()?;
            let span = Span::new(expr_span(&lhs).start, expr_span(&rhs).end);
            let _ = op_span;
            lhs = Expr::Binary {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
                span,
            };
        }
        Ok(lhs)
    }

    fn parse_term(&mut self) -> ParseResult<Expr> {
        let mut lhs = self.parse_unary()?;
        loop {
            let op = match self.current().kind {
                TokenKind::Star => BinOp::Mul,
                TokenKind::Slash => BinOp::Div,
                _ => break,
            };
            self.advance();
            let rhs = self.parse_unary()?;
            let span = Span::new(expr_span(&lhs).start, expr_span(&rhs).end);
            lhs = Expr::Binary {
                op,
                lhs: Box::new(lhs),
                rhs: Box::new(rhs),
                span,
            };
        }
        Ok(lhs)
    }

    fn parse_unary(&mut self) -> ParseResult<Expr> {
        if matches!(self.current().kind, TokenKind::Minus) {
            let start = self.current().span.start;
            self.advance();
            let inner = self.parse_unary()?;
            let end = expr_span(&inner).end;
            return Ok(Expr::UnaryNeg(Box::new(inner), Span::new(start, end)));
        }
        self.parse_primary()
    }

    fn parse_primary(&mut self) -> ParseResult<Expr> {
        let tok = self.current().clone();
        match tok.kind {
            TokenKind::Number(n) => {
                self.advance();
                Ok(Expr::Number(n, tok.span))
            }
            TokenKind::Ident(name) => {
                self.advance();
                Ok(Expr::Var(name, tok.span))
            }
            TokenKind::LParen => {
                self.advance();
                let inner = self.parse_expr()?;
                match &self.current().kind {
                    TokenKind::RParen => {
                        self.advance();
                        Ok(inner)
                    }
                    _ => Err(ParseError {
                        message: "expected `)`".into(),
                        span: self.current().span,
                    }),
                }
            }
            TokenKind::Eof => Err(ParseError {
                message: "unexpected end of input".into(),
                span: tok.span,
            }),
            _ => Err(ParseError {
                message: "expected number, identifier, or `(`".into(),
                span: tok.span,
            }),
        }
    }
}

fn expr_span(expr: &Expr) -> Span {
    match expr {
        Expr::Number(_, s) | Expr::Var(_, s) | Expr::UnaryNeg(_, s) => *s,
        Expr::Binary { span, .. } => *span,
    }
}

/// Pretty-print an AST with explicit parentheses so structure is unambiguous.
pub fn format_expr(expr: &Expr) -> String {
    match expr {
        Expr::Number(n, _) => {
            if n.fract() == 0.0 && n.abs() < 1e15 {
                format!("{}", *n as i64)
            } else {
                format!("{n}")
            }
        }
        Expr::Var(name, _) => name.clone(),
        Expr::UnaryNeg(inner, _) => format!("(-{})", format_expr(inner)),
        Expr::Binary { op, lhs, rhs, .. } => {
            format!("({} {} {})", format_expr(lhs), op, format_expr(rhs))
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn parse(s: &str) -> Expr {
        Parser::parse_expression(s).unwrap_or_else(|e| panic!("{e}"))
    }

    #[test]
    fn precedence_and_associativity() {
        let e = parse("1+2*3-4/x");
        assert_eq!(format_expr(&e), "((1 + (2 * 3)) - (4 / x))");
    }

    #[test]
    fn unary_and_parens() {
        let e = parse("--(a+b)*-2.5");
        assert_eq!(format_expr(&e), "((-( -(a + b))) * (-2.5))");
    }

    #[test]
    fn rejects_trailing_junk() {
        assert!(Parser::parse_expression("1+2)").is_err());
        assert!(Parser::parse_expression("").is_err());
        assert!(Parser::parse_expression("1.").is_err());
    }
}

#!/usr/bin/env bash
#
# pg-backup-s3.sh — Dump a PostgreSQL database, compress it, upload to S3,
# and rotate (delete) old backups beyond a retention count.
#
# All configuration can be provided via environment variables; sane defaults
# are defined below. Exits non-zero on any failure.

set -Eeuo pipefail
IFS=$'\n\t'

###############################################################################
# Configuration (env-overridable)
###############################################################################
PGHOST="${PGHOST:-localhost}"
PGPORT="${PGPORT:-5432}"
PGDATABASE="${PGDATABASE:?PGDATABASE must be set}"
PGUSER="${PGUSER:-postgres}"

export PGPASSWORD="${PGPASSWORD:-}"
# Prefer .pgpass if no password supplied
[ -z "$PGPASSWORD" ] && export PGPASSFILE="${PGPASSFILE:-$HOME/.pgpass}"

AWS_REGION="${AWS_REGION:-us-east-1}"
S3_BUCKET="${S3_BUCKET:?S3_BUCKET must be set}"
S3_PREFIX="${S3_PREFIX:-backups/${PGDATABASE}}"
RETENTION_DAYS="${RETENTION_DAYS:-30}"

WORK_DIR="${WORK_DIR:-$(mktemp -d /tmp/pgbackup.XXXXXX)}"
LOCK_FILE="${LOCK_FILE:-/var/lock/pg-backup-s3.lock}"
LOG_FILE="${LOG_FILE:-/var/log/pg-backup-s3.log}"

AWS_CLI="${AWS_CLI:-aws}"
PG_DUMP="${PG_DUMP:-pg_dump}"

TIMESTAMP="$(date -u +%Y%m%dT%H%M%SZ)"
BACKUP_NAME="${PGDATABASE}_${TIMESTAMP}.sql.gz"
LOCAL_FILE="${WORK_DIR}/${BACKUP_NAME}"
S3_URI="s3://${S3_BUCKET}/${S3_PREFIX}/${BACKUP_NAME}"

###############################################################################
# Logging, error handling, cleanup
###############################################################################
log() {
    printf '%s [%s] %s\n' "$(date -u +%Y-%m-%dT%H:%M:%SZ)" "$1" "$2" \
        | tee -a "$LOG_FILE" >&2
}
info()  { log INFO  "$1"; }
error() { log ERROR "$1"; }

cleanup() {
    rm -f "$LOCK_FILE"
    rm -rf "$WORK_DIR"
}
trap cleanup EXIT

on_error() {
    local exit_code=$? line_no=${1:-?}
    error "Script failed at line ${line_no} (exit code ${exit_code})."
    # Attempt to remove partial upload so orphans aren't counted in rotation.
    if [ -n "${S3_URI:-}" ]; then
        "$AWS_CLI" s3 rm "$S3_URI" --region "$AWS_REGION" >/dev/null 2>&1 || true
    fi
    exit "$exit_code"
}
trap 'on_error $LINENO' ERR

###############################################################################
# Pre-flight checks
###############################################################################
command -v "$AWS_CLI"  >/dev/null 2>&1 || { error "aws CLI not found.";        exit 127; }
command -v "$PG_DUMP"  >/dev/null 2>&1 || { error "pg_dump not found.";        exit 127; }
command -v gzip        >/dev/null 2>&1 || { error "gzip not found.";           exit 127; }

# Prevent concurrent runs
exec 9>"$LOCK_FILE"
if ! flock -n 9; then
    error "Another backup is already running (lock: ${LOCK_FILE})."
    exit 1
fi

info "Starting backup of '${PGDATABASE}' to '${S3_URI}'."

###############################################################################
# 1. Dump and compress
###############################################################################
info "Running pg_dump..."
if ! "$PG_DUMP" \
        --no-owner --no-privileges \
        --host="$PGHOST" --port="$PGPORT" \
        --username="$PGUSER" --dbname="$PGDATABASE" \
        | gzip -9 > "$LOCAL_FILE"; then
    error "pg_dump failed (credentials, connectivity, or pg_dump version mismatch)."
    exit 2
fi

# Sanity check: a valid dump should never be empty
if [ ! -s "$LOCAL_FILE" ]; then
    error "Backup file is empty; aborting."
    exit 2
fi

local_size=$(stat -c%s "$LOCAL_FILE" 2>/dev/null || stat -f%z "$LOCAL_FILE")
info "Dump complete (${local_size} bytes)."

###############################################################################
# 2. Upload to S3
###############################################################################
info "Uploading to ${S3_URI}..."
if ! "$AWS_CLI" s3 cp "$LOCAL_FILE" "$S3_URI" \
        --region "$AWS_REGION" \
        --storage-class STANDARD_IA \
        --only-show-errors; then
    error "S3 upload failed."
    exit 3
fi

# Verify upload by comparing sizes
remote_size=$("$AWS_CLI" s3 ls "s3://${S3_BUCKET}/${S3_PREFIX}/" --recursive \
    --region "$AWS_REGION" \
    | awk -v f="$BACKUP_NAME" '$4 == f {print $3}')
if [ "$remote_size" != "$local_size" ]; then
    error "Size mismatch after upload (local=${local_size}, remote=${remote_size:-missing})."
    exit 3
fi
info "Upload verified."

rm -f "$LOCAL_FILE"

###############################################################################
# 3. Rotate old backups (delete objects older than RETENTION_DAYS)
###############################################################################
info "Rotating backups older than ${RETENTION_DAYS} days..."
cutoff_epoch=$(date -u -d "-${RETENTION_DAYS} days" +%s 2>/dev/null \
            || date -u -v-"${RETENTION_DAYS}"d +%s)

old_objects=$("$AWS_CLI" s3api list-objects-v2 \
        --region "$AWS_REGION" \
        --bucket "$S3_BUCKET" \
        --prefix "${S3_PREFIX}/" \
        --query "Contents[?LastModified < '${cutoff_date:-}'].[Key, LastModified]" \
        --output text 2>/dev/null \
    || echo "")

# Fallback cross-platform approach: compare dates shell-side
while IFS=$'\t' read -r key last_mod; do
    [ -z "$key" ] && continue
    obj_epoch=$(date -u -d "$last_mod" +%s 2>/dev/null \
             || date -u -j -f "%Y-%m-%dT%H:%M:%SZ" "$last_mod" +%s 2>/dev/null || echo 0)
    if [ "$obj_epoch" -lt "$cutoff_epoch" ]; then
        if "$AWS_CLI" s3 rm "s3://${S3_BUCKET}/${key}" \
                --region "$AWS_REGION" --only-show-errors; then
            info "Deleted old backup: ${key}"
        else
            error "Failed to delete ${key} (continuing)."
        fi
    fi
done < <("$AWS_CLI" s3api list-objects-v2 \
            --region "$AWS_REGION" \
            --bucket "$S3_BUCKET" \
            --prefix "${S3_PREFIX}/" \
            --query 'Contents[].[Key, LastModified]' \
            --output text 2>/dev/null | grep -v '^$' || true)

info "Rotation complete."
info "Backup finished successfully: ${S3_URI}"
exit 0

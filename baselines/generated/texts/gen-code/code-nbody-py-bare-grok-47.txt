"""N-body gravitational simulation with a leapfrog (velocity Verlet) integrator.

Positions and velocities are stored as arrays of shape (n, 3). Accelerations
are computed with a vectorized all-pairs Newtonian force and a Plummer
softening length so close encounters remain finite. The leapfrog scheme is
symplectic and second-order, which keeps the total energy bounded over long
integrations rather than drifting secularly.
"""

from __future__ import annotations

import argparse
from dataclasses import dataclass
from typing import Optional, Tuple

import numpy as np

G_DEFAULT = 1.0


def gravitational_acceleration(
    positions: np.ndarray,
    masses: np.ndarray,
    g: float = G_DEFAULT,
    softening: float = 1e-3,
) -> np.ndarray:
    """Return accelerations of shape (n, 3) from Newtonian gravity.

    a_i = G * sum_{j != i} m_j * (r_j - r_i) / (|r_j - r_i|^2 + eps^2)^(3/2)
    """
    pos = np.asarray(positions, dtype=np.float64)
    mass = np.asarray(masses, dtype=np.float64)
    if pos.ndim != 2 or pos.shape[1] != 3:
        raise ValueError("positions must have shape (n, 3)")
    if mass.shape != (pos.shape[0],):
        raise ValueError("masses must have shape (n,)")
    if softening < 0.0:
        raise ValueError("softening must be non-negative")

    delta = pos[np.newaxis, :, :] - pos[:, np.newaxis, :]  # (i, j, 3) = r_j - r_i
    dist2 = np.sum(delta * delta, axis=-1) + softening * softening
    np.fill_diagonal(dist2, np.inf)  # exclude self-interaction
    inv_r3 = dist2 ** -1.5
    return g * np.sum(delta * (mass[np.newaxis, :, np.newaxis] * inv_r3[:, :, np.newaxis]), axis=1)


def kinetic_energy(velocities: np.ndarray, masses: np.ndarray) -> float:
    v2 = np.sum(np.asarray(velocities, dtype=np.float64) ** 2, axis=1)
    return 0.5 * float(np.dot(masses, v2))


def potential_energy(
    positions: np.ndarray,
    masses: np.ndarray,
    g: float = G_DEFAULT,
    softening: float = 1e-3,
) -> float:
    """Softened pairwise potential, counted once per pair: -G m_i m_j / sqrt(r^2+eps^2)."""
    pos = np.asarray(positions, dtype=np.float64)
    mass = np.asarray(masses, dtype=np.float64)
    delta = pos[np.newaxis, :, :] - pos[:, np.newaxis, :]
    dist = np.sqrt(np.sum(delta * delta, axis=-1) + softening * softening)
    np.fill_diagonal(dist, np.inf)
    # Upper triangle only so each pair is counted once.
    iu = np.triu_indices(pos.shape[0], k=1)
    return float(-g * np.sum(mass[iu[0]] * mass[iu[1]] / dist[iu]))


def total_energy(
    positions: np.ndarray,
    velocities: np.ndarray,
    masses: np.ndarray,
    g: float = G_DEFAULT,
    softening: float = 1e-3,
) -> float:
    return kinetic_energy(velocities, masses) + potential_energy(positions, masses, g, softening)


@dataclass
class Snapshot:
    time: float
    positions: np.ndarray
    velocities: np.ndarray
    energy: float


class NBodyLeapfrog:
    """Fixed-step leapfrog (kick-drift-kick) N-body integrator."""

    def __init__(
        self,
        positions: np.ndarray,
        velocities: np.ndarray,
        masses: np.ndarray,
        dt: float,
        g: float = G_DEFAULT,
        softening: float = 1e-3,
    ) -> None:
        self.positions = np.array(positions, dtype=np.float64, copy=True)
        self.velocities = np.array(velocities, dtype=np.float64, copy=True)
        self.masses = np.array(masses, dtype=np.float64, copy=True)
        if self.positions.shape != self.velocities.shape or self.positions.shape[1] != 3:
            raise ValueError("positions and velocities must both have shape (n, 3)")
        if self.masses.shape != (self.positions.shape[0],):
            raise ValueError("masses must have shape (n,)")
        if not np.all(self.masses > 0.0):
            raise ValueError("masses must be positive")
        if dt <= 0.0:
            raise ValueError("dt must be positive")
        self.dt = float(dt)
        self.g = float(g)
        self.softening = float(softening)
        self.time = 0.0
        self.accelerations = gravitational_acceleration(
            self.positions, self.masses, self.g, self.softening
        )

    @property
    def n(self) -> int:
        return int(self.masses.shape[0])

    def energy(self) -> float:
        return total_energy(self.positions, self.velocities, self.masses, self.g, self.softening)

    def center_of_mass(self) -> Tuple[np.ndarray, np.ndarray]:
        total = float(np.sum(self.masses))
        r = np.sum(self.masses[:, np.newaxis] * self.positions, axis=0) / total
        v = np.sum(self.masses[:, np.newaxis] * self.velocities, axis=0) / total
        return r, v

    def step(self) -> None:
        """Advance one leapfrog step: half kick, full drift, half kick."""
        half = 0.5 * self.dt
        self.velocities += self.accelerations * half
        self.positions += self.velocities * self.dt
        self.accelerations = gravitational_acceleration(
            self.positions, self.masses, self.g, self.softening
        )
        self.velocities += self.accelerations * half
        self.time += self.dt

    def run(
        self,
        n_steps: int,
        record_every: int = 1,
    ) -> list[Snapshot]:
        if n_steps < 0:
            raise ValueError("n_steps must be non-negative")
        if record_every < 1:
            raise ValueError("record_every must be >= 1")
        history = [self.snapshot()]
        for i in range(1, n_steps + 1):
            self.step()
            if i % record_every == 0 or i == n_steps:
                history.append(self.snapshot())
        return history

    def snapshot(self) -> Snapshot:
        return Snapshot(
            time=self.time,
            positions=self.positions.copy(),
            velocities=self.velocities.copy(),
            energy=self.energy(),
        )


def figure_eight_initial_conditions() -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Chenciner–Montgomery figure-eight, G = 1, equal masses.

    Initial data from the classic periodic three-body solution (period ~ 6.3259).
    """
    masses = np.ones(3, dtype=np.float64)
    positions = np.array(
        [
            [0.97000436, -0.24308753, 0.0],
            [-0.97000436, 0.24308753, 0.0],
            [0.0, 0.0, 0.0],
        ],
        dtype=np.float64,
    )
    v = np.array([-0.93240737, -0.86473146, 0.0], dtype=np.float64)
    velocities = np.array([-0.5 * v, -0.5 * v, v], dtype=np.float64)
    return positions, velocities, masses


def random_cluster(
    n: int,
    seed: int = 0,
    box: float = 1.0,
    velocity_scale: float = 0.1,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    rng = np.random.default_rng(seed)
    positions = rng.uniform(-box, box, size=(n, 3))
    velocities = rng.normal(0.0, velocity_scale, size=(n, 3))
    masses = rng.uniform(0.5, 1.5, size=n)
    # Remove bulk motion so the cluster stays near the origin.
    velocities -= np.sum(masses[:, np.newaxis] * velocities, axis=0) / np.sum(masses)
    return positions, velocities, masses


def integrate(
    positions: np.ndarray,
    velocities: np.ndarray,
    masses: np.ndarray,
    dt: float,
    n_steps: int,
    g: float = G_DEFAULT,
    softening: float = 1e-3,
    record_every: int = 1,
) -> Tuple[NBodyLeapfrog, list[Snapshot]]:
    sim = NBodyLeapfrog(positions, velocities, masses, dt=dt, g=g, softening=softening)
    history = sim.run(n_steps, record_every=record_every)
    return sim, history


def _demo(argv: Optional[list[str]] = None) -> None:
    parser = argparse.ArgumentParser(description="Leapfrog N-body demo")
    parser.add_argument("--steps", type=int, default=2000)
    parser.add_argument("--dt", type=float, default=1e-3)
    parser.add_argument("--softening", type=float, default=1e-4)
    parser.add_argument("--system", choices=("figure8", "cluster"), default="figure8")
    parser.add_argument("--n", type=int, default=16, help="bodies for the random cluster")
    args = parser.parse_args(argv)

    if args.system == "figure8":
        pos, vel, mass = figure_eight_initial_conditions()
    else:
        pos, vel, mass = random_cluster(args.n)

    sim, history = integrate(
        pos, vel, mass, dt=args.dt, n_steps=args.steps, softening=args.softening, record_every=max(args.steps // 10, 1)
    )
    e0 = history[0].energy
    e1 = history[-1].energy
    r_cm, v_cm = sim.center_of_mass()
    print(f"bodies={sim.n} steps={args.steps} dt={args.dt} t={sim.time:.6g}")
    print(f"E0={e0:.8e} E={e1:.8e} rel={(e1 - e0) / e0:.3e}")
    print(f"|r_cm|={np.linalg.norm(r_cm):.3e} |v_cm|={np.linalg.norm(v_cm):.3e}")


if __name__ == "__main__":
    _demo()
